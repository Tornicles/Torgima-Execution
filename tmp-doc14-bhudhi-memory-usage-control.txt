TASK 1 — BHUDHI MEMORY SERVICE
// BhudhiMemoryService.ts
// Reads from and writes to the existing bhudhi_memory table (DOC 02 schema)
 
// MEMORY TYPES (allowed):
// user_goal_summary: cleaned goal and purpose — expires 30 days after blueprint update
// recent_task_context: last 3 task types completed — expires 7 days
// friction_summary: current dominant friction type — expires 3 days
// capability_summary: current proven capability level — expires 14 days
// recovery_context: last inactivity type and severity — expires 7 days
// pattern_summary: current active behavior patterns (no raw data) — expires 7 days
 
// WHAT MEMORY IS NOT:
// Not a raw dump of feedback records
// Not a copy of signal scores
// Not private reflection notes
// Not credential or session data
 
export class BhudhiMemoryService {
  // Write or update a memory entry
  async upsert(userId: string, blueprintId: string, memoryType: string,
    content: Record&lt;string, unknown&gt;, expiryDays: number): Promise&lt;void&gt;
 
  // Load memory for a specific trigger (returns only what that trigger needs)
  async loadForTrigger(userId: string, blueprintId: string,
    triggerType: string): Promise&lt;Record&lt;string, unknown&gt;&gt;
 
  // MEMORY INJECTION RULES per trigger:
  // TASK_DRAFTING: inject user_goal_summary + recent_task_context
  // RECOVERY_COPY: inject recovery_context + friction_summary
  // ACADEMY_SUPPORT: inject capability_summary
  // STATE_EXPLANATION: inject pattern_summary (summary only, not raw)
  // GOAL_STRUCTURING: no memory (structuring raw input — memory would contaminate)
  // BLOCKER_CLASSIFICATION: no memory (must classify from current input only)
  // PROOF_RELEVANCE_CHECK: no memory
 
  // MEMORY DECAY:
  // Run nightly: delete bhudhi_memory WHERE expires_at &lt; NOW()
  async runDecayCleanup(): Promise&lt;void&gt;
 
  // CONTEXT MINIMIZATION:
  // Memory content must be summarized to &lt;= 200 chars per entry
  // before being injected into context packet
  // Never inject raw JSON blobs as memory
  private summarizeMemory(content: Record&lt;string, unknown&gt;): string
}

TASK 3 — CONTEXT BUILDER WITH MEMORY
// Extend ContextBuilder.ts to inject memory when appropriate
 
async function build(
  input: Record&lt;string, unknown&gt;,
  requiredFields: string[],
  forbiddenFields: string[],
  userId: string,
  blueprintId: string,
  triggerType: string,
): Promise&lt;ContextPacket&gt; {
 
  // 1. Filter input to required fields only (existing behavior)
  const filtered = pickAllowedFields(input, requiredFields, forbiddenFields);
 
  // 2. Load relevant memory for this trigger type
  const memory = await BhudhiMemoryService.loadForTrigger(
    userId, blueprintId, triggerType
  );
 
  // 3. Inject memory as summarized context (not raw)
  return { ...filtered, memory_context: memory, stateVersion: input.stateVersion };
}
