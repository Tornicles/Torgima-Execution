// FailureClassificationEngine.ts
// Triggered by: failure_classification_requested event
// Sources: FAILED feedback, BLOCKED feedback, rejected proof, outcome mismatch, pattern
 
export async function classifyFailure(request: FailureClassificationRequest) {
  // 1. Load all evidence
  const evidence = await loadFailureEvidence({
    userId: request.userId,
    blueprintId: request.blueprintId,
    orderId: request.orderId,
    sourceType: request.sourceType,
    sourceId: request.sourceId,
  });
 
  // evidence includes: feedback record, order context, task generation log,
  // capability profile, condition record, behavior patterns, outcome validations,
  // inactivity records, false progress flags
 
  // 2. Score all 10 failure categories
  const scores = await RootCauseScorer.scoreAll(evidence);
 
  // 3. Determine primary category (highest score) and secondary categories
  const primaryCategory = scores[0].category;
  const secondaryCategories = scores.slice(1, 3).filter(s => s.score >= 30).map(s => s.category);
 
  // 4. Determine confidence based on evidence richness
  const confidence = evidence.feedbackReflectionLength > 50 ? &apos;HIGH&apos;
    : evidence.feedbackReflectionLength > 20 ? &apos;MEDIUM&apos; : &apos;LOW&apos;;
 
  // 5. Determine severity
  const severity = await determineSeverity(primaryCategory, evidence);
 
  // 6. Check for repeat failure
  const repeatFailure = await RepeatFailureDetector.check(
    request.userId, request.blueprintId, primaryCategory
  );
 
  // 7. Write DecisionTrace
  await DecisionTraceService.write({
    decisionType: &apos;SYSTEM_CORRECTION&apos;,
    decisionKey: `failure_classification:${request.sourceId}`,
    selectedOption: primaryCategory,
    optionsConsidered: scores.map(s => s.category),
    rationaleSummary: `Primary: ${primaryCategory} (score: ${scores[0].score}). Evidence: ${evidence.summary}`,
    confidence, stateVersion: request.stateVersion,
  });
 
  // 8. Write BehaviorEvent if pattern-relevant
  if ([&apos;CAPABILITY_FAILURE&apos;,&apos;CAPACITY_FAILURE&apos;,&apos;MOTIVATION_FRICTION&apos;].includes(primaryCategory)) {
    await writeBehaviorEvent(request.userId, request.blueprintId, primaryCategory, request.sourceId);
  }
 
  // 9. Route correction for MEDIUM/HIGH/CRITICAL severity
  if (severity !== &apos;LOW&apos;) {
    await CorrectionRouter.route(primaryCategory, request, repeatFailure);
  }
 
  return { primaryCategory, secondaryCategories, confidence, severity, repeatFailure };
}

// RepeatFailureDetector.ts
// Checks if this failure category has appeared before in current window
 
// REPEAT RULES:
// Same primary category appears 2+ times in 7-day window:
//   → escalate confidence from MEDIUM to HIGH
//   → escalate severity by 1 level (LOW→MEDIUM, MEDIUM→HIGH, HIGH→CRITICAL)
 
// Same ORDER fails 2+ times (same order_id, different feedback records):
//   → do NOT regenerate same task unchanged
//   → emit task_variation_requested to TaskGenerationEngine
//   → classify as CLARITY_FAILURE even if original category was different
 
// Same PROOF TYPE rejected 3+ times:
//   → flag for UI/proof contract review
//   → emit system_friction_detected (proof UX may be broken)
