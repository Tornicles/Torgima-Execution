Task Generation Engine Specification
Converting Blueprint Position + Signal State Into the Next Correct Task
Phase 1 â€” Individual System

1. Document Purpose

This document defines the Task Generation Engine â€” the system that produces the next correct task for the user to execute. The engine reads the blueprint execution position, the enforcement package constraints, the user's capability profile, relevant knowledge objects, and generates tasks with complete content: instruction, expected output, definition of done, proof requirements, and why it matters.

Every task the user ever sees in Torgima is produced by this engine. It never produces a task that violates the enforcement package. It always produces tasks that are positioned correctly in the blueprint chain. If Bhudhi is available, it enriches the task content. If Bhudhi is unavailable, the engine falls back to template-based generation.

The task generation engine has one absolute rule: it must always produce at least one task when called. If it cannot generate a normal execution task, it generates a recovery, clarification, or academy task. It never returns empty. The ContinuousTaskGuarantee acts as the final safety net after this engine.

2.2 Context Loading Implementation
async function loadGenerationContext(
  blueprint_id: string,
  user_id:      string,
  enforcement_package_id: string,
  generation_trigger: string,
  generation_mode:    string,
): Promise&lt;TaskGenerationContext&gt; {

  const [blueprint, pkg, capability] = await Promise.all([
    prisma.blueprint.findUnique({
      where: { id: blueprint_id },
      include: {
        current_plan:  { select: { id:true, name:true } },
        current_phase: { select: { id:true, name:true, status:true } },
      },
    }),
    prisma.enforcementPackage.findFirst({
      where: { id: enforcement_package_id },
    }),
    prisma.capabilityProfile.findFirst({
      where: { user_id },
      select: { level:true, consistency:true, execution_quality:true, current_streak_days:true },
    }),
  ]);

  if (!blueprint) throw new TorgimaError("BLUEPRINT_NOT_FOUND", "Blueprint not found");
  if (!pkg) throw new TorgimaError("NO_ENFORCEMENT_PACKAGE", "Enforcement package not found");

  // Find current target and phase
  const current_target = await prisma.target.findFirst({
    where: { blueprint_id, status: { in: ["active","on_track","at_risk"] } },
    orderBy: { order_index: "asc" },
  });

  const current_phase = blueprint.current_phase_id
    ? await prisma.phase.findUnique({ where: { id: blueprint.current_phase_id } })
    : null;

  // Find current project
  const current_project = await prisma.project.findFirst({
    where: {
      blueprint_id,
      phase_id: current_phase?.id,
      status: { in: ["active","pending"] },
    },
    orderBy: { order_index: "asc" },
  });

  // Find next action to execute
  const next_action = current_project
    ? await prisma.action.findFirst({
        where: {
          project_id: current_project.id,
          status: "pending",
          type: { in: (pkg.allowed_actions as string[]) },
        },
        orderBy: { order_index: "asc" },
      })
    : null;

  // Load relevant knowledge objects
  const knowledge_objects = await prisma.knowledgeObject.findMany({
    where: {
      AND: [
        { OR: [{ blueprint_id }, { is_global: true }] },
        { OR: [
          { domain: blueprint.current_domain as any ?? undefined },
          { domain: null },
        ]},
        { applicable_level: { in: [capability?.level ?? "beginner","all"] } },
        { active: true },
      ],
    },
    take: 5,
  });

  // Load recent task content hashes for duplicate detection
  const recent_orders = await prisma.order.findMany({
    where: { blueprint_id, user_id, created_at: { gte: new Date(Date.now() - 7*86400000) } },
    select: { content_hash: true },
    take: 20,
  });

  const task_limits = pkg.task_limits as { min: number; max: number };

  return {
    blueprint:   { id: blueprint.id, goal: blueprint.goal, purpose: blueprint.purpose ?? undefined, current_domain: blueprint.current_domain ?? undefined },
    plan:        blueprint.current_plan ?? null,
    season:      null, // loaded from blueprint.current_season if needed
    target:      current_target ?? null,
    phase:       current_phase ?? null,
    project:     current_project ?? null,
    next_action: next_action ?? null,
    enforcement: {
      task_limits,
      difficulty_cap:      pkg.difficulty_cap,
      allowed_actions:     pkg.allowed_actions as string[],
      blocked_actions:     pkg.blocked_actions as string[],
      ai_mode:             pkg.ai_mode ?? "command",
      recovery_required:   (pkg.override_flags as any)?.force_recovery ?? false,
      feature_permissions: pkg.feature_permissions as Record&lt;string, boolean&gt;,
    },
    capability: {
      level:             capability?.level ?? "beginner",
      consistency:       capability?.consistency ?? 0,
      execution_quality: capability?.execution_quality ?? 0,
      current_streak:    capability?.current_streak_days ?? 0,
    },
    knowledge_objects,
    recent_task_hashes: recent_orders.map(o=&gt;o.content_hash).filter(Boolean) as string[],
    generation_trigger,
    generation_mode,
    enforcement_package_id,
  };
}

3.3 Task Type Selection Function
async function selectTaskType(
  ctx: TaskGenerationContext,
  blueprint_id: string,
  user_id: string,
): Promise&lt;string[]&gt; {
  const types: string[] = [];

  // Recovery mode â€” only recovery tasks
  if (ctx.generation_mode === "recovery") {
    return ["recovery"];
  }

  // Recovery required by escalation
  if (ctx.enforcement.recovery_required) {
    types.push("recovery");
  }

  // Phase checkpoint due
  if (ctx.phase?.status === "gated" || await isPhaseCheckpointDue(blueprint_id, ctx.phase?.id)) {
    types.push("phase_checkpoint");
    return types; // Phase checkpoint takes priority over execution
  }

  // Weekly review due (Monday within 48h of week start)
  if (await isWeeklyReviewDue(blueprint_id, user_id)) {
    types.push("review");
  }

  // Active blocker
  const blocked = await prisma.order.findFirst({
    where: { blueprint_id, user_id, status: "blocked" },
  });
  if (blocked) {
    types.push("blocker_resolution");
  }

  // Academy gap
  const academy = await prisma.academyTask.findFirst({
    where: { blueprint_id, user_id, status: "pending", blocks_execution: true },
  });
  if (academy) {
    types.push("academy");
    return types; // Academy that blocks execution takes priority
  }

  // Execution â€” normal path
  if (ctx.next_action) {
    types.push("execution");
  } else {
    // No action available â€” clarification needed
    types.push("clarification");
  }

  return types;
}

// Helpers
async function isPhaseCheckpointDue(blueprint_id: string, phase_id?: string): Promise&lt;boolean&gt; {
  if (!phase_id) return false;
  const phase = await prisma.phase.findUnique({ where: { id: phase_id } });
  return phase?.status === "gated";
}

async function isWeeklyReviewDue(blueprint_id: string, user_id: string): Promise&lt;boolean&gt; {
  const lastReview = await prisma.weeklyReview.findFirst({
    where: { blueprint_id, user_id },
    orderBy: { created_at: "desc" },
  });
  if (!lastReview) return false;
  const daysSince = (Date.now() - lastReview.created_at.getTime()) / 86400000;
  return daysSince &gt;= 7;
}


4. Difficulty Selection

The difficulty of each task is selected within the bounds set by the enforcement package's difficulty_cap. The engine uses the user's capability level, the action type, the blueprint position, and generation mode to determine the ideal difficulty.

4.1 Difficulty Selection Logic
const DIFFICULTY_ORDER = ["very_easy","easy","moderate","hard","very_hard"];

function selectDifficulty(params: {
  difficulty_cap:    string;
  capability_level:  string;
  action_type?:      string;
  generation_mode:   string;
  current_streak:    number;
  is_first_in_phase: boolean;
}): string {
  const cap_idx = DIFFICULTY_ORDER.indexOf(params.difficulty_cap);

  // Recovery and initial modes always start easy
  if (params.generation_mode === "recovery" || params.generation_mode === "initial") {
    return DIFFICULTY_ORDER[Math.min(1, cap_idx)] ?? "easy"; // easy or cap, whichever lower
  }

  // First task in a new phase starts one step below cap
  if (params.is_first_in_phase) {
    return DIFFICULTY_ORDER[Math.max(0, cap_idx - 1)] ?? "easy";
  }

  // High-impact action types (analyze, improve, standardize) use full cap
  const hard_action_types = ["analyze","improve","standardize","build"];
  if (params.action_type && hard_action_types.includes(params.action_type)) {
    return params.difficulty_cap; // Full cap
  }

  // Streak bonus: after 7+ consecutive days, allow full cap
  if (params.current_streak &gt;= 7) {
    return params.difficulty_cap;
  }

  // Default: one level below cap for normal execution
  const default_idx = Math.max(0, cap_idx - 1);
  return DIFFICULTY_ORDER[default_idx] ?? "moderate";
}

5. Task Value Scoring

Every generated task receives a value score (0â€“10) that represents its strategic importance. This score is stored on the order and is used by the ContinuousTaskGuarantee and the dashboard to surface the highest-value tasks first.

5.1 Value Score Calculation
function calculateTaskValueScore(params: {
  action_type:      string;
  blueprint_position: {
    is_blocking_next_phase: boolean;
    target_progress: number;    // 0-1
    phase_name:      string;
  };
  difficulty:       string;
  task_type:        string;
  generation_mode:  string;
  capability_level: string;
  time_estimate_minutes?: number;
  is_critical_path: boolean;
  knowledge_required: boolean;
  is_blocking_blocker: boolean;
  target_at_risk: boolean;
  streak_at_risk: boolean;
  conditions: {
  completion_rate: number; weighted_score: number;
  };
  }): number {

  let score = 5.0; // Baseline

  // Action type contribution (0â€“2)
  const action_scores: Record&lt;string, number&gt; = {
    create:     1.0, replicate: 0.8, analyze:    1.5,
    improve:    1.5, standardize: 1.8, recover:  0.5,
    test:       1.2, publish:    1.8, validate:  1.2,
    measure:    1.3, build:      1.8,
  };
  score += action_scores[params.action_type] ?? 1.0;

  // Blueprint position (0â€“1.5)
  if (params.blueprint_position.is_blocking_next_phase) score += 1.5;
  if (params.blueprint_position.target_progress &gt;= 0.8) score += 0.5;

  // Criticality flags (0â€“2)
  if (params.is_critical_path) score += 1.0;
  if (params.target_at_risk) score += 1.0;
  if (params.is_blocking_blocker) score += 0.5;
  if (params.streak_at_risk) score += 0.5;

  // Difficulty contribution (0â€“1)
  const diff_scores: Record&lt;string, number&gt; = {
    very_easy: 0.0, easy: 0.2, moderate: 0.4, hard: 0.7, very_hard: 1.0,
  };
  score += diff_scores[params.difficulty] ?? 0.4;

  // Knowledge required adds value (+0.5)
  if (params.knowledge_required) score += 0.5;

  // Cap at 10
  return Math.min(score, 10.0);
}

Titles must start with a verb: Write, Build, Analyze, Test, Review, Create, Define, Launch.
Instructions must be specific to the user's goal, current project, and action â€” not generic advice.
Definition of done must be a binary condition â€” either it is met or it is not. No partial credit in the definition.

Expected output must name a real artifact: a document, a list, a message sent, a file, a measurement, a decision made.
Proof is required when: action type is publish, validate, or build; difficulty is hard or very_hard; task type is phase_checkpoint.

7.3 Bhudhi Call with Timeout and Fallback
// Bhudhi integration with 5-second timeout and fallback
async function generateWithBhudhi(
  ctx: TaskGenerationContext,
  task_type: string,
  difficulty: string,
): Promise&lt;{ content: TaskContent; used_bhudhi: boolean }&gt; {

  const ai_enabled = await prisma.featureFlag.findUnique({
    where: { feature_name: "ai_enabled" },
    select: { enabled: true },
  });

  if (!ai_enabled?.enabled) {
    return { content: generateTemplate(ctx, task_type, difficulty), used_bhudhi: false };
  }

  const input: BhudhiTaskGenerationInput = buildBhudhiInput(ctx, task_type, difficulty);

  try {
    const response = await Promise.race([
      callBhudhi(input),
      new Promise&lt;never&gt;((_, reject) =&gt;
        setTimeout(() =&gt; reject(new Error("Bhudhi timeout")), 5000)
      ),
    ]);

    const output = response as BhudhiTaskGenerationOutput;

    // Validate Bhudhi output
    if (!validateBhudhiOutput(output)) {
      logger.warn("[TaskEngine] Bhudhi output invalid â€” using template", { task_type });
      return { content: generateTemplate(ctx, task_type, difficulty), used_bhudhi: false };
    }

    return { content: output.data, used_bhudhi: true };

  } catch (err) {
    logger.warn("[TaskEngine] Bhudhi failed â€” using template fallback", {
      error: (err as Error).message, task_type,
    });
    return { content: generateTemplate(ctx, task_type, difficulty), used_bhudhi: false };
  }
}

async function callBhudhi(input: BhudhiTaskGenerationInput): Promise&lt;BhudhiTaskGenerationOutput&gt; {
  // Calls src/bhudhi/BhudhiCore.ts â€” defined in Document 12
  const { BhudhiCore } = await import("@/bhudhi/BhudhiCore");
  const core = new BhudhiCore();
  return core.execute({ trigger_type: "task_generation", payload: input }) as Promise&lt;BhudhiTaskGenerationOutput&gt;;
}


8. Template-Based Task Generation

9. Content Hash and Duplicate Detection

Every generated task gets a content_hash based on its key content fields. This hash is checked against recent tasks before storing to prevent duplicate tasks from appearing in the user's queue.

import { createHash } from "crypto";

function generateContentHash(content: TaskContent, blueprint_position: {
  action_id?: string; project_id?: string; phase_id?: string;
}): string {
  const hashInput = [
    content.title,
    content.instruction.substring(0, 100), // First 100 chars
    blueprint_position.action_id ?? "",
    blueprint_position.project_id ?? "",
  ].join("|");

  return createHash("sha256").update(hashInput).digest("hex").substring(0, 16);
}

function isDuplicateTask(hash: string, recent_hashes: string[]): boolean {
  return recent_hashes.includes(hash);
}

// src/engines/task/TaskGenerationEngine.ts
import { prisma } from "@/lib/prisma";
import { logger } from "@/lib/logger";
import { TaskGenerationError } from "@/lib/errors";

export interface GenerationInput {
  blueprint_id:           string;
  user_id:                string;
  enforcement_package_id: string;
  generation_trigger:     string;
  generation_mode:        string;
}

export interface GenerationResult {
  orders_generated: number;
  order_ids:        string[];
  used_bhudhi:      boolean;
  used_fallback:    boolean;
}

export class TaskGenerationEngine {

  async generate(input: GenerationInput): Promise&lt;GenerationResult&gt; {
    const { blueprint_id, user_id, enforcement_package_id,
      generation_trigger, generation_mode } = input;

    logger.info("[TaskEngine] Generating tasks", {
      blueprint_id, trigger: generation_trigger, mode: generation_mode,
    });

    // 1. Load full generation context
    const ctx = await loadGenerationContext(
      blueprint_id, user_id, enforcement_package_id,
      generation_trigger, generation_mode,
    );

    // 2. Determine how many tasks to generate
    const task_count = this.selectTaskCount(ctx.enforcement.task_limits, ctx.capability.current_streak);

    // 3. Determine task types needed
    const task_types = await selectTaskType(ctx, blueprint_id, user_id);

    // 4. Generate each task
    const generated_orders: string[] = [];
    let used_bhudhi = false;
    let used_fallback = false;

    for (let i = 0; i &lt; Math.min(task_count, task_types.length + 1); i++) {
      const task_type = task_types[i] ?? "execution";

      // Select difficulty for this task
      const is_first_in_phase = i === 0 && generation_mode === "initial";
      const difficulty = selectDifficulty({
        difficulty_cap:    ctx.enforcement.difficulty_cap,
        capability_level:  ctx.capability.level,
        action_type:       ctx.next_action?.type,
        generation_mode,
        current_streak:    ctx.capability.current_streak,
        is_first_in_phase,
      });

      // Generate content (Bhudhi or template)
      const { content, used_bhudhi: bhudhi_used } = await generateWithBhudhi(
        ctx, task_type, difficulty,
      );
      if (bhudhi_used) used_bhudhi = true;
      else used_fallback = true;

      // Generate content hash for duplicate detection
      const hash = generateContentHash(content, {
        action_id:  ctx.next_action?.id,
        project_id: ctx.project?.id,
        phase_id:   ctx.phase?.id,
      });

      // Skip if duplicate
      if (isDuplicateTask(hash, ctx.recent_task_hashes)) {
        logger.info("[TaskEngine] Duplicate task skipped", { hash });
        used_fallback = true;
        continue;
      }

      // Calculate value score
      const value_score = calculateTaskValueScore({
        action_type: task_type === "execution" ? (ctx.next_action?.type ?? "create") : task_type,
        blueprint_position: {
          is_blocking_next_phase: ctx.phase?.status === "gated",
          target_progress: ctx.target ? ctx.target.progress / 100 : 0,
          phase_name: ctx.phase?.name ?? "initiation",
        },
        difficulty,
        task_type,
        generation_mode,
        capability_level: ctx.capability.level,
        time_estimate_minutes: content.time_estimate_minutes,
        is_critical_path: ctx.next_action?.knowledge_category === "sop",
        knowledge_required: ctx.knowledge_objects.length &gt; 0,
        is_blocking_blocker: task_type === "blocker_resolution",
        target_at_risk: ctx.target?.progress !== undefined && ctx.target.progress &lt; 30,
        streak_at_risk: ctx.capability.current_streak === 1,
        conditions: { completion_rate: 0.5, weighted_score: 5 },
      });

      // Store order in transaction
      const order = await prisma.$transaction(async (tx) =&gt; {
        const o = await tx.order.create({
          data: {
            blueprint_id,
            user_id,
            action_id:              ctx.next_action?.id,
            linked_plan_id:         ctx.plan?.id,
            linked_target_id:       ctx.target?.id,
            linked_phase_id:        ctx.phase?.id,
            linked_project_id:      ctx.project?.id,
            task_type:              task_type as any,
            title:                  content.title,
            instruction:            content.instruction,
            expected_output:        content.expected_output,
            definition_of_done:     content.definition_of_done,
            why_it_matters:         content.why_it_matters,
            time_estimate_minutes:  content.time_estimate_minutes,
            proof_required:         content.proof_required,
            proof_type:             content.proof_type as any,
            proof_description:      content.proof_description,
            difficulty:             difficulty as any,
            priority:               Math.round(value_score),
            value_score,
            domain:                 ctx.blueprint.current_domain as any,
            content_hash:           hash,
            condition_at_generation: (await prisma.conditionRecord.findFirst({
              where: { blueprint_id, is_current: true }, select: { condition: true }
            }))?.condition ?? "unknown",
            domain_at_generation:    ctx.blueprint.current_domain ?? "unknown",
            enforcement_package_id,
            generation_reason: `${generation_trigger} â€” ${generation_mode} â€” ${task_type}`,
            status: "pending",
          },
        });

        // Link knowledge objects
        if (ctx.knowledge_objects.length &gt; 0) {
          await Promise.all(ctx.knowledge_objects.slice(0,3).map(k =&gt;
            tx.orderKnowledgeLink.create({
              data: { order_id: o.id, knowledge_id: k.id, injection_type: "implicit" },
            }).catch(() =&gt; null) // Ignore duplicate link errors
          ));
        }

        return o;
      });

      generated_orders.push(order.id);
      ctx.recent_task_hashes.push(hash); // Prevent same hash in this batch
    }

    // 5. Store task generation log
    await prisma.taskGenerationLog.create({
      data: {
        blueprint_id, user_id,
        generation_trigger: generation_trigger as any,
        generation_mode:    generation_mode as any,
        condition_at_generation: (await prisma.conditionRecord.findFirst({
          where: { blueprint_id, is_current: true }, select: { condition: true },
        }))?.condition ?? "unknown",
        enforcement_package_id,
        capability_level:   ctx.capability.level,
        domain_at_generation: ctx.blueprint.current_domain ?? "unknown",
        input_json:  { task_types, task_count, blueprint_position: { plan: ctx.plan?.name, phase: ctx.phase?.name } },
        orders_generated: generated_orders.length,
        order_ids:        generated_orders,
        generation_notes: `${used_bhudhi ? "Bhudhi" : "Template"} generation. Types: ${task_types.join(",")}`,
        used_fallback,
        ai_used: used_bhudhi,
      },
    });

    logger.info("[TaskEngine] Generation complete", {
      blueprint_id, orders_generated: generated_orders.length, used_bhudhi,
    });

    return {
      orders_generated: generated_orders.length,
      order_ids:        generated_orders,
      used_bhudhi,
      used_fallback,
    };
  }

  private selectTaskCount(
    task_limits: { min: number; max: number },
    current_streak: number,
  ): number {
    // Start at minimum, add based on streak
    const streak_bonus = Math.floor(current_streak / 5); // +1 per 5-day streak
    const target = Math.min(task_limits.min + streak_bonus, task_limits.max);
    return target;
  }
}


11. Blueprint Position Advancement

After a task is generated for an action, the engine may need to mark that action as active. When a task tied to an action is completed, the action advances to complete and the next action becomes active. This progression is handled by the BlueprintEngine, not the TaskGenerationEngine â€” the task engine only reads position, never writes it.

The TaskGenerationEngine ONLY reads blueprint position. It never modifies plan, target, phase, project, or action status. Blueprint position is advanced by the BlueprintEngine, which is triggered by phase_checkpoint events and feedback completion events.

11.1 Position Fallback When Blueprint Position Is Empty
If the engine cannot find a current project or action â€” because all projects in the phase are complete â€” it generates a clarification task and triggers a blueprint position check event.

// No active action found â€” check if all projects complete

  task_types = ["clarification"];
}


12. Generation Modes

The generation_mode field determines how the engine approaches task generation. Each mode has specific behaviour differences.

Mode
Behaviour
initial
First tasks for a newly launched blueprint. Always starts with easy difficulty. Task count = min (1â€“2). Generates only execution or clarification tasks.
post_feedback
Standard mode. Called after every feedback submission. Full task type selection. Full difficulty selection. Normal task count.
daily
Called by daily cycle worker. Ensures user always has tasks at start of day. Only generates if active_task_count &lt; task_limits.min.
recovery
Forced recovery mode. Only generates recovery tasks. Max 1 task. Always very_easy difficulty.
condition_change
Called when condition changes by 2+ levels. Regenerates task set. Clears old pending tasks that no longer fit the new difficulty_cap.

12.1 Condition Change Mode â€” Task Cleanup
// When condition_change mode is triggered:
// Old pending tasks that exceed the new difficulty_cap must be cancelled
// New tasks are generated within the new constraints

async function clearInvalidTasks(blueprint_id: string, user_id: string, difficulty_cap: string) {
  const DIFFICULTY_ORDER = ["very_easy","easy","moderate","hard","very_hard"];
  const cap_idx = DIFFICULTY_ORDER.indexOf(difficulty_cap);

  // Find pending tasks that exceed the new cap
  const invalid_tasks = await prisma.order.findMany({
    where: {
      blueprint_id, user_id, status: "pending",
      difficulty: { in: DIFFICULTY_ORDER.slice(cap_idx + 1) as any[] },
    },
    select: { id: true },
  });

  if (invalid_tasks.length &gt; 0) {
    await prisma.order.updateMany({
      where: { id: { in: invalid_tasks.map(t =&gt; t.id) } },
      data:  { status: "cancelled" },
    });

    logger.info("[TaskEngine] Cancelled tasks exceeding new difficulty cap", {
      blueprint_id, cancelled: invalid_tasks.length, new_cap: difficulty_cap,
    });
  }
}
