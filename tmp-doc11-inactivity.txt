// InactivityDetectionEngine.ts
// Called by scheduled worker daily at midnight (existing cron worker)
// Also called when: order has been pending > threshold with no activity
 
// DETECTION RULES â€” classify as inactive when:
 
// SHORT_ABSENCE (1-2 days):
//   Last feedback OR last order start > 24 hours ago
//   AND blueprint is active
//   AND at least 1 pending or active order exists
//   Action: no intervention yet â€” log only, watch window
 
// MODERATE_ABSENCE (3-5 days):
//   Last feedback OR activity > 72 hours ago
//   Action: emit inactivity_detected event, generate gentle nudge
//   InactivityRecord.severity = 'MODERATE'
 
// EXTENDED_ABSENCE (6-13 days):
//   Last feedback > 6 days ago
//   Action: emit inactivity_detected, generate recovery task
//   InactivityRecord.severity = 'HIGH'
//   Reduce enforcement to minimum â€” user needs easy re-entry
 
// ABANDONMENT_RISK (14+ days):
//   Last feedback > 14 days ago
//   Action: emit abandonment_risk event, generate restart task
//   InactivityRecord.severity = 'CRITICAL'
//   Condition forced to URGENCY minimum for re-entry
 
// DETECTION METHOD:
async function detectInactivity(userId: string, blueprintId: string) {
  const lastFeedback = await getLastFeedbackTimestamp(userId, blueprintId);
  const lastOrderActivity = await getLastOrderActivityTimestamp(userId, blueprintId);
  const lastActivity = max(lastFeedback, lastOrderActivity);
  const hoursInactive = (Date.now() - lastActivity.getTime()) / 3_600_000;
 
  if (hoursInactive < 24) return null;  // not inactive
 
  const severity = hoursInactive < 72 ? 'MODERATE'
    : hoursInactive < 144 ? 'HIGH'
    : 'CRITICAL';
 
  // Check for existing open InactivityRecord â€” do not duplicate
  const existing = await getOpenInactivityRecord(userId, blueprintId);
  if (existing && existing.severity === severity) return existing;  // already detected
 
  // Create new InactivityRecord
  return await createInactivityRecord({ userId, blueprintId, severity, hoursInactive });
}
