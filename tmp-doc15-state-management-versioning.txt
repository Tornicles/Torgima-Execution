TASK 2 — STATEMUTATION SERVICE
// StateMutationService.ts
// All critical state mutations go through this service
 
export async function applyMutation&lt;T&gt;(
  blueprintId: string,
  expectedStateVersion: number,
  mutationType: string,
  mutationFn: (tx: PrismaTransaction) =&gt; Promise&lt;T&gt;,
): Promise&lt;StateMutationResult&lt;T&gt;&gt; {
 
  return await prisma.$transaction(async (tx) =&gt; {
    // 1. Load current state version with lock
    const blueprint = await tx.blueprint.findUnique({
      where: { id: blueprintId },
      select: { state_version: true, id: true },
    });
 
    // 2. Check for stale state
    if (blueprint.state_version !== expectedStateVersion) {
      return {
        ok: false,
        result: &apos;REJECTED_STALE&apos;,
        previousStateVersion: blueprint.state_version,
        error: &apos;STATE_VERSION_MISMATCH&apos;,
      };
    }
 
    // 3. Apply the mutation
    const data = await mutationFn(tx);
 
    // 4. Increment state version
    const newVersion = expectedStateVersion + 1;
    await tx.blueprint.update({
      where: { id: blueprintId },
      data: { state_version: newVersion },
    });
 
    // 5. Write BlueprintMutation record
    await tx.blueprintMutation.create({
      data: {
        blueprint_id: blueprintId,
        mutation_type: mutationType,
        previous_version: expectedStateVersion,
        new_version: newVersion,
        applied_at: new Date(),
      }
    });
 
    return { ok: true, result: &apos;APPLIED&apos;,
      previousStateVersion: expectedStateVersion,
      newStateVersion: newVersion, data };
  });
}
