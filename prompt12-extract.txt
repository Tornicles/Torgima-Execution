export default async function DashboardPage() {
  const session = await getServerSession()
  const userId = session.user.id

  // Parallel fetch everything
  const [snapshot, activeOrder, morningReport, pendingProposals] = await Promise.all([
    getDashboardSnapshot(userId),
    getActiveOrder(userId),
    getLatestMorningReport(userId),
    getPendingProposals(await getActiveBlueprintId(userId)),
  ])

  // Ensure active order (guarantee engine)
  const order = activeOrder ?? await ensureActiveOrder(userId)

  return (
    <DashboardLayout>
      {/* Primary Execution Row */}
      <ExecutionRow>
        <ActiveOrderCard order={order} />
        <ConditionBadge condition={snapshot.executionBrief.condition} />
      </ExecutionRow>

      {/* Intelligence Surface - 2 column grid */}
      <IntelligenceGrid>
        <StrategicOutlookPanel outlook={snapshot.strategicOutlook} />
        <BehavioralInsightCard insight={snapshot.behavioralInsight} />
        <SignalTrendPanel signals={snapshot.signalNarrative} />
        <BlueprintHealthPanel health={snapshot.blueprintHealth} />
      </IntelligenceGrid>

      {/* Morning Intelligence Report - full width */}
      {morningReport && <MorningReport report={morningReport} />}

      {/* Governance Action Center - conditional */}
      {pendingProposals.length > 0 && (
        <GovernanceActionCenter proposals={pendingProposals} />
      )}
    </DashboardLayout>
  )
}

// LAYOUT RULES:
// - Primary Execution Row: always at top, always visible
// - Intelligence Grid: responsive 2-column on desktop, 1-column mobile
// - Each panel: expandable/collapsible, user preference stored in settings
// - Governance Action Center: only shown when proposals pending
// - Morning Report: always below intelligence grid (not above it)
