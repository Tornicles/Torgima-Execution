import { prisma } from "@/lib/prisma";
import { logger } from "@/lib/logger";
import { writeEvent, EVENT_TYPES } from "@/services/EventService";
import { Prisma } from "@prisma/client";

// Phase 1 domain activation status (from DOC_40)
const DOMAIN_PHASE1_STATUS: Record<string, "always_active" | "active" | "limited" | "disabled"> = {
  genesis:     "always_active",
  market:      "active",
  delivery:    "active",
  continuity:  "active",
  operations:  "limited",
  growth:      "disabled",
  people:      "disabled",
};

// Assign domain at blueprint launch based on goal context
export function resolveInitialDomain(params: {
  goal: string;
  domainHint?: string | null;
  validationVerdict?: string | null;
  selectedPath?: string | null;
}): string {
  const hint = params.domainHint?.toLowerCase();

  // If user provided a domain hint and it is active
  if (hint && DOMAIN_PHASE1_STATUS[hint] && DOMAIN_PHASE1_STATUS[hint] !== "disabled") {
    return hint;
  }

  // Default: Market if validation happened (validated demand)
  // Delivery if goal implies execution/fulfillment is ready
  // Genesis fallback if nothing else applies
  if (params.validationVerdict === "proceed" || params.selectedPath) {
    // After validation, move into Market by default
    return "market";
  }

  return "genesis";
}

// Transition to a new domain — enforces Phase 1 rules
export async function transitionDomain(params: {
  blueprintId: string;
  userId: string;
  targetDomain: string;
  trigger: string;
  triggerEventId?: string;
  notes?: string;
  tx?: Prisma.TransactionClient;
}): Promise<{ transitioned: boolean; domain: string; reason?: string }> {
  const db = params.tx ?? prisma;

  const status = DOMAIN_PHASE1_STATUS[params.targetDomain];

  if (!status) {
    logger.warn("[DomainEngine] Unknown domain transition target", { domain: params.targetDomain });
    return { transitioned: false, domain: params.targetDomain, reason: "unknown_domain" };
  }

  if (status === "disabled") {
    // Log blocked transition — do not change domain
    await db.event.create({
      data: {
        event_type: "domain_transition_blocked",
        entity_type: "blueprint",
        entity_id: params.blueprintId,
        user_id: params.userId,
        blueprint_id: params.blueprintId,
        payload: {
          target_domain: params.targetDomain,
          reason: "domain_disabled_phase1",
          trigger: params.trigger,
        },
      },
    });
    logger.info("[DomainEngine] Domain transition blocked — disabled in Phase 1", {
      blueprintId: params.blueprintId,
      targetDomain: params.targetDomain,
    });
    return { transitioned: false, domain: params.targetDomain, reason: "disabled_phase1" };
  }

  // Get current domain
  const current = await db.domainHistory.findFirst({
    where: { blueprint_id: params.blueprintId, is_current: true },
    orderBy: { entered_at: "desc" },
  });

  if (current?.domain === params.targetDomain) {
    return { transitioned: false, domain: params.targetDomain, reason: "already_in_domain" };
  }

  // Exit current domain
  if (current) {
    await db.domainHistory.update({
      where: { id: current.id },
      data: { is_current: false, exited_at: new Date() },
    });
  }

  // Enter new domain
  await db.domainHistory.create({
    data: {
      blueprint_id: params.blueprintId,
      user_id: params.userId,
      domain: params.targetDomain as any,
      previous_domain: current?.domain ?? null,
      transition_trigger: params.trigger as any,
      trigger_event_id: params.triggerEventId,
      transition_notes: params.notes,
      is_current: true,
    },
  });

  // Update blueprint current_domain
  await db.blueprint.update({
    where: { id: params.blueprintId },
    data: { current_domain: params.targetDomain as any },
  });

  logger.info("[DomainEngine] Domain transition completed", {
    blueprintId: params.blueprintId,
    from: current?.domain,
    to: params.targetDomain,
    trigger: params.trigger,
  });

  return { transitioned: true, domain: params.targetDomain };
}

// Evaluate whether domain should transition based on signals/conditions
export async function evaluateDomainTransition(params: {
  blueprintId: string;
  userId: string;
  currentDomain: string;
  conditionName: string;
  conditionDurationDays: number;
  phaseCompleted?: boolean;
}): Promise<string | null> {
  // Continuity trigger: sustained Danger 10+ days
  if (
    params.conditionDurationDays >= 10 &&
    (params.conditionName === "danger") &&
    params.currentDomain !== "continuity"
  ) {
    return "continuity";
  }

  // Phase complete in Genesis → move to Market
  if (params.phaseCompleted && params.currentDomain === "genesis") {
    return "market";
  }

  // Phase complete in Market with good signal → move to Delivery
  if (
    params.phaseCompleted &&
    params.currentDomain === "market" &&
    ["stability", "power", "affluence"].includes(params.conditionName)
  ) {
    return "delivery";
  }

  // All other transitions are handled by StrategyEngine (Prompt 12)
  return null;
}

// ── BUILD BLUEPRINT CHAIN ────────────────────────────────────────────────────
// Called after validation — builds the full Plan → Target → Phase → Project → Action chain

export async function buildBlueprintChain(params: {
  blueprintId: string;
  userId: string;
  chain: BlueprintChainInput;
}): Promise<void> {
  const blueprint = await prisma.blueprint.findFirst({
    where: { id: params.blueprintId, user_id: params.userId },
  });
  if (!blueprint) throw new TorgimaError("NOT_FOUND", "Blueprint not found", false);

  await prisma.$transaction(async (tx) => {
    const { chain } = params;
    const plans = chain.pathSummary?.plans ?? defaultPlans();
    const targets = chain.pathSummary?.targets ?? defaultTargets(chain.goal);

    let planIndex = 0;
    for (const planInput of plans) {
      const plan = await tx.plan.create({
        data: {
          blueprint_id: params.blueprintId,
          name: planInput.name,
          description: planInput.description,
          order_index: planIndex++,
          status: "pending",
        },
      });

      // Create one Season per plan (Ignition for first, Foundation for rest)
      await tx.season.create({
        data: {
          blueprint_id: params.blueprintId,
          plan_id: plan.id,
          name: planIndex === 1 ? "ignition" : "foundation",
          order_index: planIndex - 1,
          status: "pending",
        },
      });

      // Create targets for this plan
      for (const targetInput of targets) {
        const target = await tx.target.create({
          data: {
            blueprint_id: params.blueprintId,
            plan_id: plan.id,
            owner_id: params.userId, // Individual mode — user owns all targets
            name: targetInput.name,
            description: targetInput.description,
            type: targetInput.type,
            metric: targetInput.metric,
            target_value: targetInput.targetValue,
            unit: targetInput.unit,
            order_index: 0,
            status: "pending",
          },
        });

        // Create 3 phases per target: Initiation → Development → Completion
        const phaseNames: Array<"initiation" | "development" | "completion"> = [
          "initiation",
          "development",
          "completion",
        ];

        for (let phaseIdx = 0; phaseIdx < phaseNames.length; phaseIdx++) {
          const phase = await tx.phase.create({
            data: {
              blueprint_id: params.blueprintId,
              target_id: target.id,
              plan_id: plan.id,
              name: phaseNames[phaseIdx],
              order_index: phaseIdx,
              gate_criteria: buildGateCriteria(phaseNames[phaseIdx]),
              status: phaseIdx === 0 ? "active" : "pending",
              locked: phaseIdx > 0, // Only first phase is unlocked
            },
          });

          // Create projects for the initiation phase only (others built when phase gates open)
          if (phaseIdx === 0) {
            const projectTypes: Array<"setup" | "execution"> = ["setup", "execution"];
            for (let projIdx = 0; projIdx < projectTypes.length; projIdx++) {
              const project = await tx.project.create({
                data: {
                  blueprint_id: params.blueprintId,
                  phase_id: phase.id,
                  target_id: target.id,
                  owner_id: params.userId,
                  name: buildProjectName(projectTypes[projIdx], targetInput.name),
                  type: projectTypes[projIdx],
                  order_index: projIdx,
                  status: projIdx === 0 ? "active" : "pending",
                },
              });

              // Create 2–3 actions per project
              const actionTypes = buildActionTypes(projectTypes[projIdx]);
              for (let actIdx = 0; actIdx < actionTypes.length; actIdx++) {
                await tx.action.create({
                  data: {
                    blueprint_id: params.blueprintId,
                    project_id: project.id,
                    target_id: target.id,
                    name: buildActionName(actionTypes[actIdx], targetInput.name),
                    type: actionTypes[actIdx] as any,
                    knowledge_category: resolveKnowledgeCategory(actionTypes[actIdx]),
                    order_index: actIdx,
                    status: actIdx === 0 ? "pending" : "pending",
                  },
                });
              }
            }
          }
        }
      }

      // Set current_plan to first plan
      if (planIndex === 1) {
        await tx.blueprint.update({
          where: { id: params.blueprintId },
          data: { current_plan_id: plan.id },
        });
      }
    }

    // Write BlueprintVersion for chain creation
    await tx.blueprintVersion.create({
      data: {
        blueprint_id: params.blueprintId,
        version: 1,
        summary: "Initial blueprint chain created from validation path",
        structure_snapshot: { plans: plans.length, targets: targets.length } as any,
        source: "USER",
        triggered_by: params.userId,
      },
    });

    // Increment state version
    await tx.blueprint.update({
      where: { id: params.blueprintId },
      data: {
        status: "path_selection",
        state_version: { increment: 1 },
      },
    });
  });

  logger.info("[BlueprintEngine] Blueprint chain built", { blueprintId: params.blueprintId });
}

// ── ACTIVATE BLUEPRINT (LAUNCH) ──────────────────────────────────────────────

export async function activateBlueprint(params: {
  blueprintId: string;
  userId: string;
}): Promise<void> {
  const blueprint = await prisma.blueprint.findFirst({
    where: { id: params.blueprintId, user_id: params.userId },
    include: {
      plans: { orderBy: { order_index: "asc" }, take: 1 },
    },
  });

  if (!blueprint) throw new TorgimaError("NOT_FOUND", "Blueprint not found", false);
  if (blueprint.status !== "path_selection" && blueprint.status !== "draft") {
    throw new TorgimaError("BLUEPRINT_NOT_ACTIVE", `Blueprint status is ${blueprint.status}`, false);
  }

  const now = new Date();
  const windowStart = new Date(now.getTime() - 7 * 24 * 3600000);

  // Resolve initial domain
  const initialDomain = resolveInitialDomain({
    goal: blueprint.goal,
    domainHint: blueprint.domain_hint,
    validationVerdict: "proceed",
    selectedPath: "current",
  });

  await prisma.$transaction(async (tx) => {
    // 1. Activate blueprint
    await tx.blueprint.update({
      where: { id: params.blueprintId },
      data: {
        status: "active",
        active: true,
        state_version: { increment: 1 },
      },
    });

    // 2. Set first plan as current
    if (blueprint.plans[0]) {
      // Find first active phase in first plan
      const firstPhase = await tx.phase.findFirst({
        where: { blueprint_id: params.blueprintId, plan_id: blueprint.plans[0].id },
        orderBy: { order_index: "asc" },
      });
      await tx.blueprint.update({
        where: { id: params.blueprintId },
        data: {
          current_plan_id: blueprint.plans[0].id,
          current_phase_id: firstPhase?.id ?? null,
        },
      });
      // Mark first plan active
      await tx.plan.update({
        where: { id: blueprint.plans[0].id },
        data: { status: "active" },
      });
    }

    // 3. Initial signal snapshot — nothingness (insufficient data)
    const signal = await tx.signalSnapshot.create({
      data: {
        blueprint_id: params.blueprintId,
        user_id: params.userId,
        window_start: windowStart,
        window_end: now,
        completion_rate: 0,
        weighted_score: 0,
        momentum: 0,
        trend: "stable",
        confidence: "insufficient_data",
        assigned_condition: "nothingness",
        is_current: true,
      },
    });

    // 4. Initial condition — nothingness
    const condition = await tx.conditionRecord.create({
      data: {
        blueprint_id: params.blueprintId,
        user_id: params.userId,
        condition: "nothingness",
        signal_snapshot_id: signal.id,
        duration_days: 0,
        escalation_level: 0,
        recovery_required: false,
        is_current: true,
        entered_at: now,
      },
    });

    // 5. Initial enforcement package — nothingness defaults
    await tx.enforcementPackage.create({
      data: {
        blueprint_id: params.blueprintId,
        condition_id: condition.id,
        signal_snapshot_id: signal.id,
        condition: "nothingness",
        task_limits: { min: 1, max: 2 },
        difficulty_cap: "easy",
        ai_mode: "command",
        ui_mode: "starter",
        feature_permissions: {},
        is_current: true,
      },
    });

    // 6. Initial state memory
    await tx.stateMemory.create({
      data: {
        blueprint_id: params.blueprintId,
        user_id: params.userId,
        condition: "nothingness",
        entered_at: now,
        is_current: true,
        duration_days: 0,
        escalation_level: 0,
      },
    });

    // 7. Initial domain history
    await tx.domainHistory.create({
      data: {
        blueprint_id: params.blueprintId,
        user_id: params.userId,
        domain: initialDomain as any,
        transition_trigger: "blueprint_launched",
        is_current: true,
        entered_at: now,
      },
    });
    await tx.blueprint.update({
      where: { id: params.blueprintId },
      data: { current_domain: initialDomain as any },
    });

    // 8. Emit blueprint_launched event (triggers TaskGenerationWorker)
    await writeEvent(
      {
        eventType: EVENT_TYPES.BLUEPRINT_LAUNCHED,
        entityType: "blueprint",
        entityId: params.blueprintId,
        userId: params.userId,
        blueprintId: params.blueprintId,
        payload: {
          generation_mode: "initial",
          generation_trigger: "blueprint_launched",
          domain: initialDomain,
        },
        idempotencyKey: `blueprint_launched:${params.blueprintId}`,
      },
      tx
    );
  });

  logger.info("[BlueprintEngine] Blueprint activated", {
    blueprintId: params.blueprintId,
    userId: params.userId,
    domain: initialDomain,
  });
}

// ── ADVANCE BLUEPRINT POSITION ────────────────────────────────────────────────
// Called after phase gate passes — moves to next phase or next plan

export async function advanceBlueprintPosition(params: {
  blueprintId: string;
  userId: string;
  completedPhaseId: string;
}): Promise<{ advanced: boolean; newPhaseId?: string; newPlanId?: string }> {
  const blueprint = await prisma.blueprint.findFirst({
    where: { id: params.blueprintId, user_id: params.userId, active: true },
    include: {
      plans: { orderBy: { order_index: "asc" } },
    },
  });

  if (!blueprint) throw new TorgimaError("NOT_FOUND", "Blueprint not found", false);

  const completedPhase = await prisma.phase.findUnique({
    where: { id: params.completedPhaseId },
  });

  if (!completedPhase || completedPhase.blueprint_id !== params.blueprintId) {
    throw new TorgimaError("NOT_FOUND", "Phase not found", false);
  }

  // Find next phase in same target
  const nextPhase = await prisma.phase.findFirst({
    where: {
      blueprint_id: params.blueprintId,
      target_id: completedPhase.target_id,
      order_index: { gt: completedPhase.order_index },
      status: { not: "complete" },
    },
    orderBy: { order_index: "asc" },
  });

  await prisma.$transaction(async (tx) => {
    // Mark completed phase
    await tx.phase.update({
      where: { id: params.completedPhaseId },
      data: { status: "complete", completed_at: new Date(), gate_passed: true },
    });

    if (nextPhase) {
      // Unlock and activate next phase
      await tx.phase.update({
        where: { id: nextPhase.id },
        data: { status: "active", locked: false },
      });
      await tx.blueprint.update({
        where: { id: params.blueprintId },
        data: {
          current_phase_id: nextPhase.id,
          state_version: { increment: 1 },
        },
      });

      // Build projects for newly unlocked phase (if not already built)
      const existingProjects = await tx.project.count({
        where: { blueprint_id: params.blueprintId, phase_id: nextPhase.id },
      });

      if (existingProjects === 0) {
        await buildPhaseProjects(params.blueprintId, params.userId, nextPhase.id, tx);
      }
    }

    // Write BlueprintMutation
    await tx.blueprintMutation.create({
      data: {
        blueprint_id: params.blueprintId,
        mutation_type: "COMPLETE",
        layer: "phase",
        entity_id: params.completedPhaseId,
        change_summary: `Phase ${completedPhase.name} completed. Advanced to ${nextPhase?.name ?? "end"}`,
        approved: true,
        approved_at: new Date(),
      },
    });
  });

  return {
    advanced: true,
    newPhaseId: nextPhase?.id,
  };
}

// ── PAUSE BLUEPRINT ──────────────────────────────────────────────────────────

export async function pauseBlueprint(params: {
  blueprintId: string;
  userId: string;
  reason?: string;
}): Promise<void> {
  const blueprint = await prisma.blueprint.findFirst({
    where: { id: params.blueprintId, user_id: params.userId, active: true },
  });
  if (!blueprint) throw new TorgimaError("NOT_FOUND", "Blueprint not found or not active", false);
  if (blueprint.status === "paused") throw new TorgimaError("BLUEPRINT_PAUSED", "Blueprint already paused", false);

  await prisma.$transaction(async (tx) => {
    await tx.blueprint.update({
      where: { id: params.blueprintId },
      data: {
        status: "paused",
        paused_at: new Date(),
        pause_reason: params.reason,
        state_version: { increment: 1 },
      },
    });

    await writeEvent(
      {
        eventType: EVENT_TYPES.BLUEPRINT_PAUSED,
        entityType: "blueprint",
        entityId: params.blueprintId,
        userId: params.userId,
        blueprintId: params.blueprintId,
        payload: { reason: params.reason },
        idempotencyKey: `blueprint_paused:${params.blueprintId}:${Date.now()}`,
      },
      tx
    );
  });

  logger.info("[BlueprintEngine] Blueprint paused", { blueprintId: params.blueprintId });
}

// ── RESUME BLUEPRINT ─────────────────────────────────────────────────────────

export async function resumeBlueprint(params: {
  blueprintId: string;
  userId: string;
}): Promise<void> {
  const blueprint = await prisma.blueprint.findFirst({
    where: { id: params.blueprintId, user_id: params.userId },
  });
  if (!blueprint) throw new TorgimaError("NOT_FOUND", "Blueprint not found", false);
  if (blueprint.status !== "paused") throw new TorgimaError("BLUEPRINT_NOT_ACTIVE", "Blueprint is not paused", false);

  const pausedAt = blueprint.paused_at;
  const pauseGapDays = pausedAt
    ? Math.floor((Date.now() - pausedAt.getTime()) / (24 * 3600 * 1000))
    : 0;

  const resumeMode =
    pauseGapDays <= 2 ? "normal" :
    pauseGapDays <= 7 ? "check_in" :
    pauseGapDays <= 30 ? "recovery" : "restart";

  await prisma.$transaction(async (tx) => {
    await tx.blueprint.update({
      where: { id: params.blueprintId },
      data: {
        status: "active",
        active: true,
        paused_at: null,
        pause_reason: null,
        state_version: { increment: 1 },
      },
    });

    // Trigger signal recalculation + task generation with appropriate mode
    await writeEvent(
      {
        eventType: EVENT_TYPES.BLUEPRINT_RESUMED,
        entityType: "blueprint",
        entityId: params.blueprintId,
        userId: params.userId,
        blueprintId: params.blueprintId,
        payload: { resume_mode: resumeMode, pause_gap_days: pauseGapDays },
        idempotencyKey: `blueprint_resumed:${params.blueprintId}:${Date.now()}`,
      },
      tx
    );

    await writeEvent(
      {
        eventType: EVENT_TYPES.TASK_GENERATION_REQUESTED,
        entityType: "blueprint",
        entityId: params.blueprintId,
        userId: params.userId,
        blueprintId: params.blueprintId,
        payload: {
          generation_mode: resumeMode === "normal" ? "post_feedback" : "recovery",
          generation_trigger: "blueprint_launched",
        },
        idempotencyKey: `task_gen_resume:${params.blueprintId}:${Date.now()}`,
      },
      tx
    );
  });

  logger.info("[BlueprintEngine] Blueprint resumed", {
    blueprintId: params.blueprintId,
    resumeMode,
    pauseGapDays,
  });
}

function buildGateCriteria(phaseName: string): Record<string, unknown> {
  const gates: Record<string, unknown> = {
    initiation:  { require_task_completion: true,  min_tasks_completed: 2,  require_reflection: false },
    development: { require_task_completion: true,  min_tasks_completed: 5,  require_proof: true },
    completion:  { require_task_completion: true,  require_outcome_validated: true, require_proof: true },
  };

function buildProjectName(type: string, targetName: string): string {
  const names: Record<string, string> = {
    setup:        `Setup: ${targetName}`,
    execution:    `Execute: ${targetName}`,
    analysis:     `Analyze: ${targetName}`,
    optimization: `Optimize: ${targetName}`,
  };
  return names[type] ?? `${type}: ${targetName}`;
}

function buildActionTypes(projectType: string): string[] {
  const actions: Record<string, string[]> = {
    setup:        ["create", "validate"],
    execution:    ["build", "test", "publish"],
    analysis:     ["analyze", "measure"],
    optimization: ["improve", "standardize"],
    recovery:     ["recover"],
  };
  return actions[projectType] ?? ["create", "analyze"];
}

function buildActionName(actionType: string, targetName: string): string {
  const verbs: Record<string, string> = {
    create: "Create",  build: "Build",   analyze: "Analyze",
    improve: "Improve", test: "Test",    validate: "Validate",
    publish: "Publish", measure: "Measure", recover: "Recover",
    standardize: "Standardize",
  };
  return `${verbs[actionType] ?? actionType}: ${targetName}`;
}

function resolveKnowledgeCategory(actionType: string): string {
  const categories: Record<string, string> = {
    create: "creation",    build: "execution",  analyze: "analysis",
    improve: "optimization", test: "validation", validate: "validation",
    publish: "delivery",   measure: "measurement", recover: "recovery",
  };
  return categories[actionType] ?? "execution";
}

async function buildPhaseProjects(
  blueprintId: string,
  userId: string,
  phaseId: string,
  tx: any
): Promise<void> {
  const phase = await tx.phase.findUnique({ where: { id: phaseId }, include: { target: true } });
  if (!phase) return;

  const projectType = phase.name === "development" ? "execution" : "analysis";
  const project = await tx.project.create({
    data: {
      blueprint_id: blueprintId,
      phase_id: phaseId,
      target_id: phase.target_id,
      owner_id: userId,
      name: buildProjectName(projectType, phase.target?.name ?? "target"),
      type: projectType as any,
      order_index: 0,
      status: "active",
    },
  });

  const actionTypes = buildActionTypes(projectType);
  for (let i = 0; i < actionTypes.length; i++) {
    await tx.action.create({
      data: {
        blueprint_id: blueprintId,
        project_id: project.id,
        target_id: phase.target_id,
        name: buildActionName(actionTypes[i], phase.target?.name ?? "target"),
        type: actionTypes[i] as any,
        knowledge_category: resolveKnowledgeCategory(actionTypes[i]),
        order_index: i,
        status: "pending",
      },
    });
  }
}
