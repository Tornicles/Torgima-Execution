import crypto from "crypto";

// Session config
const SESSION_MAX_HOURS = 24 * 30;      // 30 days absolute max
const SESSION_IDLE_HOURS = 24 * 7;      // 7 days idle expiry
const SESSION_REFRESH_INTERVAL_MS = 60 * 60 * 1000; // refresh lastSeenAt at most every 1 hour
const MAGIC_LINK_EXPIRY_MINUTES = 15;

// ── MAGIC LINK ──────────────────────────────────────────────────────────────

export async function sendMagicLink(email: string): Promise<{ sent: boolean }> {
  // Normalize email
  const normalizedEmail = email.toLowerCase().trim();

  // Generate a secure token
  const rawToken = crypto.randomBytes(32).toString("hex");
  const tokenHash = crypto.createHash("sha256").update(rawToken).digest("hex");
  const expiresAt = new Date(Date.now() + MAGIC_LINK_EXPIRY_MINUTES * 60 * 1000);

  // Upsert user (create if new, find if existing)
  const user = await prisma.user.upsert({
    where: { email: normalizedEmail },
    update: {},
    create: { email: normalizedEmail },
  });

  // Store the magic link token in sessions table with a pending status
  // We store it with a short expiry — once verified it becomes a full session
  await prisma.session.create({
    data: {
      user_id: user.id,
      token_hash: `magic:${tokenHash}`,
      expires_at: expiresAt,
    },
  });

  // Build the verify URL
  const appUrl = process.env.NEXT_PUBLIC_APP_URL ?? "http://localhost:3000";
  const verifyUrl = `${appUrl}/verify?token=${rawToken}&email=${encodeURIComponent(normalizedEmail)}`;

  // Send email via Resend
  try {
    await resend.emails.send({
      from: EMAIL_FROM,
      to: normalizedEmail,
      subject: "Sign in to Torgima",
      html: `
        <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto;">
          <h2>Sign in to Torgima</h2>
          <p>Click the link below to sign in. This link expires in ${MAGIC_LINK_EXPIRY_MINUTES} minutes.</p>
          <p><a href="${verifyUrl}" style="background: #2E75B6; color: white; padding: 12px 24px; text-decoration: none; border-radius: 4px; display: inline-block;">Sign in to Torgima</a></p>
          <p>If you didn't request this, you can safely ignore this email.</p>
          <p style="color: #888; font-size: 12px;">This link can only be used once and expires in ${MAGIC_LINK_EXPIRY_MINUTES} minutes.</p>
        </div>
      `,
      text: `Sign in to Torgima: ${verifyUrl}\n\nThis link expires in ${MAGIC_LINK_EXPIRY_MINUTES} minutes.`,
    });
  } catch (error) {
    logger.error("[AuthService] Failed to send magic link email", {
      userId: user.id,
      error: error instanceof Error ? error.message : String(error),
    });
    // Don't expose email delivery failure to prevent email enumeration
  }

  logger.info("[AuthService] Magic link sent", { userId: user.id });
  return { sent: true };
}

// ── VERIFY MAGIC LINK ───────────────────────────────────────────────────────

export async function verifyMagicLink(
  rawToken: string,
  email: string
): Promise<{ sessionToken: string; user: SafeUser }> {
  const tokenHash = crypto.createHash("sha256").update(rawToken).digest("hex");
  const normalizedEmail = email.toLowerCase().trim();

  // Find the magic link session
  const magicSession = await prisma.session.findFirst({
    where: {
      token_hash: `magic:${tokenHash}`,
      expires_at: { gt: new Date() },
      user: { email: normalizedEmail },
    },
    include: { user: true },
  });

  if (!magicSession) {
    throw new TorgimaError("UNAUTHENTICATED", "Invalid or expired magic link", false);
  }

  // Delete the magic link session (single use)
  await prisma.session.delete({ where: { id: magicSession.id } });

  // Create a real session
  const sessionRawToken = crypto.randomBytes(32).toString("hex");
  const sessionTokenHash = crypto.createHash("sha256").update(sessionRawToken).digest("hex");
  const expiresAt = new Date(Date.now() + SESSION_MAX_HOURS * 60 * 60 * 1000);

  await prisma.session.create({
    data: {
      user_id: magicSession.user_id,
      token_hash: sessionTokenHash,
      expires_at: expiresAt,
    },
  });

  logger.info("[AuthService] Magic link verified, session created", {
    userId: magicSession.user_id,
  });

  return {
    sessionToken: sessionRawToken,
    user: toSafeUser(magicSession.user),
  };
}

// ── VALIDATE SESSION ────────────────────────────────────────────────────────

export async function validateSession(sessionToken: string): Promise<AuthContext> {
  const tokenHash = crypto.createHash("sha256").update(sessionToken).digest("hex");

  const session = await prisma.session.findUnique({
    where: { token_hash: tokenHash },
    include: { user: true },
  });

  if (!session) {
    throw new TorgimaError("UNAUTHENTICATED", "Session not found", false);
  }

  if (session.expires_at < new Date()) {
    // Clean up expired session
    await prisma.session.delete({ where: { id: session.id } }).catch(() => null);
    throw new TorgimaError("UNAUTHENTICATED", "Session expired", false);
  }

  if (!session.user.active) {
    throw new TorgimaError("FORBIDDEN", "Account is inactive", false);
  }

  // Refresh lastSeenAt at most every hour to avoid excessive writes
  const lastRefresh = session.last_used_at.getTime();
  if (Date.now() - lastRefresh > SESSION_REFRESH_INTERVAL_MS) {
    await prisma.session.update({
      where: { id: session.id },
      data: { last_used_at: new Date() },
    }).catch(() => null); // Non-critical — don't fail the request
  }

  return {
    userId: session.user_id,
    sessionId: session.id,
    userMode: session.user.mode as "individual" | "team",
    userTier: session.user.tier,
    isInternal: false,
  };
}

// ── REVOKE SESSION ──────────────────────────────────────────────────────────

export async function revokeSession(sessionToken: string): Promise<void> {
  const tokenHash = crypto.createHash("sha256").update(sessionToken).digest("hex");
  await prisma.session.deleteMany({ where: { token_hash: tokenHash } });
  logger.info("[AuthService] Session revoked");
}

export async function getUser(userId: string): Promise<SafeUser> {
  const user = await prisma.user.findUnique({ where: { id: userId } });
  if (!user) throw new TorgimaError("NOT_FOUND", "User not found", false);
  return toSafeUser(user);
}

import { TorgimaError } from "@/lib/errors";
import { logger } from "@/lib/logger";
import { AuthClass, AuthContext } from "@/types/auth";

const INTERNAL_AUTH_HEADER = "x-internal-auth";
const INTERNAL_AUTH_SECRET = process.env.INTERNAL_AUTH_SECRET ?? "";
const SESSION_COOKIE_NAME = "torgima_session";

// Extract session token from cookie or Authorization header
function extractSessionToken(req: NextRequest): string | null {
  const cookieToken = req.cookies.get(SESSION_COOKIE_NAME)?.value;
  if (cookieToken) return cookieToken;

  const authHeader = req.headers.get("authorization");
  if (authHeader?.startsWith("Bearer ")) return authHeader.slice(7);

  return null;
}

// Validate internal service auth
function validateInternalAuth(req: NextRequest): AuthContext {
  const secret = req.headers.get(INTERNAL_AUTH_HEADER);
  if (!secret || secret !== INTERNAL_AUTH_SECRET) {
    throw new TorgimaError("FORBIDDEN", "Internal auth required", false);
  }
  return {
    userId: "internal",
    sessionId: "internal",
    userMode: "individual",
    userTier: "system",
    isInternal: true,
  };
}

// Main auth middleware
export async function requireAuth(
  req: NextRequest,
  authClass: AuthClass
): Promise<AuthContext> {
  if (authClass === "PUBLIC") {
    return { userId: "", sessionId: "", userMode: "individual", userTier: "", isInternal: false };
  }

  if (authClass === "INTERNAL_SERVICE" || authClass === "ADMIN_INTERNAL") {
    // Check if request origin is from browser — reject if so
    const origin = req.headers.get("origin");
    if (origin) {
      logger.error("[Auth] Browser attempted to call internal route", { origin });
      throw new TorgimaError("FORBIDDEN", "Internal route not accessible from browser", false);
    }
    return validateInternalAuth(req);
  }

  const sessionToken = extractSessionToken(req);

  if (!sessionToken) {
    if (authClass === "OPTIONAL_SESSION") {
      return { userId: "", sessionId: "", userMode: "individual", userTier: "", isInternal: false };
    }
    throw new TorgimaError("UNAUTHENTICATED", "Authentication required", false);
  }

  try {
    return await validateSession(sessionToken);
  } catch (error) {
    if (authClass === "OPTIONAL_SESSION") {
      return { userId: "", sessionId: "", userMode: "individual", userTier: "", isInternal: false };
    }
    throw error;
  }
}

// Ownership guard — call after requireAuth
export function assertOwnership(
  authContext: AuthContext,
  resourceUserId: string,
  resourceType: string
): void {
  if (authContext.isInternal) return; // Internal service bypasses ownership
  if (authContext.userId !== resourceUserId) {
    logger.error("[Auth] Ownership mismatch", {
      authUserId: authContext.userId,
      resourceUserId,
      resourceType,
    });
    // Return 404 for hidden resources to prevent enumeration
    throw new TorgimaError("NOT_FOUND", `${resourceType} not found`, false);
  }
}

  const event = await db.event.create({
    data: {
      event_type:      input.eventType,
      entity_type:     input.entityType,
      entity_id:       input.entityId,

    eventId: event.id,
    eventType: event.event_type,
    blueprintId: input.blueprintId,

  return event.id;
}

const CLAIM_TIMEOUT_MINUTES = 5;
const DISPATCH_BATCH_SIZE = 50;

export async function dispatchPendingEvents(): Promise<number> {
  const claimBefore = new Date(Date.now() - CLAIM_TIMEOUT_MINUTES * 60 * 1000);

  // Return stuck CLAIMED events to PENDING
  await prisma.event.updateMany({
    where: {
      status: "claimed",
      updated_at: { lt: claimBefore },
    },
    data: { status: "pending" },
  });

  // Claim a batch of PENDING events
  const events = await prisma.$transaction(async (tx) => {
    const pending = await tx.event.findMany({
      where: {
        status: "pending",
        process_after: { lte: new Date() },
      },
      orderBy: { created_at: "asc" },
      take: DISPATCH_BATCH_SIZE,
    });

    if (pending.length === 0) return [];

    const ids = pending.map(e => e.id);
    await tx.event.updateMany({
      where: { id: { in: ids } },
      data: { status: "claimed" },
    });

    return pending;
  });

  if (events.length === 0) return 0;

  const boss = await getBoss();
  let dispatched = 0;

  for (const event of events) {
    const queueName = EVENT_QUEUE_MAP[event.event_type];

    if (!queueName) {
      logger.warn("[EventDispatcher] No queue mapping for event type", {
        eventType: event.event_type,
        eventId: event.id,
      });
      await markEventProcessed(event.id, "noop_no_queue");
      continue;
    }

    try {
      // Check idempotency — if a job for this event already succeeded, skip
      if (event.idempotency_key) {
        const existingLog = await prisma.eventLog.findFirst({
          where: {
            event_id: event.id,
            result: "success",
          },
        });
        if (existingLog) {
          await markEventProcessed(event.id, "noop_duplicate");
          continue;
        }
      }

      await boss.send(queueName, {
        eventId: event.id,
        eventType: event.event_type,
        entityType: event.entity_type,
        entityId: event.entity_id,
        userId: event.user_id,
        blueprintId: event.blueprint_id,
        payload: event.payload,
      });

      dispatched++;
      logger.debug("[EventDispatcher] Event dispatched to queue", {
        eventId: event.id,
        queue: queueName,
      });
    } catch (error) {
      logger.error("[EventDispatcher] Failed to dispatch event", {
        eventId: event.id,
        error: error instanceof Error ? error.message : String(error),
      });
      // Return to pending so it gets retried
      await prisma.event.update({
        where: { id: event.id },
        data: { status: "pending", retry_count: { increment: 1 } },
      });
    }
  }

  return dispatched;
}

async function markEventProcessed(eventId: string, result: string): Promise<void> {
  await prisma.event.update({
    where: { id: eventId },
    data: { status: "processed" },
  });
  await prisma.eventLog.create({
    data: {
      event_id: eventId,
      result: result as any,
      processed_at: new Date(),
    },
  });
}

import { prisma } from "@/lib/prisma";
import { logger } from "@/lib/logger";

export interface WorkerJobData {
  eventId: string;
  eventType: string;
  entityType: string;
  entityId?: string;
  userId?: string;
  blueprintId?: string;
  payload: Record<string, unknown>;
}

export interface WorkerResult {
  success: boolean;
  retryable?: boolean;
  reason?: string;
}

// Standard worker handler wrapper — handles logging, retry, and dead-letter
export async function handleWorkerJob(
  job: any,
  workerName: string,
  handler: (data: WorkerJobData) => Promise<WorkerResult>
): Promise<void> {
  const data = job.data as WorkerJobData;
  const startedAt = new Date();

  logger.info(`[${workerName}] Processing`, {
    eventId: data.eventId,
    eventType: data.eventType,
    blueprintId: data.blueprintId,
  });

  try {
    // Check idempotency: has this event already been successfully processed?
    const existingSuccess = await prisma.eventLog.findFirst({
      where: { event_id: data.eventId, result: "success" },
    });

    if (existingSuccess) {
      logger.info(`[${workerName}] Noop duplicate`, { eventId: data.eventId });
      await writeEventLog(data.eventId, workerName, "noop_duplicate", startedAt);
      // Mark event processed
      await prisma.event.update({
        where: { id: data.eventId },
        data: { status: "processed" },
      }).catch(() => null);
      return;
    }

    const result = await handler(data);

    if (result.success) {
      // Mark event processed
      await prisma.event.update({
        where: { id: data.eventId },
        data: { status: "processed" },
      }).catch(() => null);
      await writeEventLog(data.eventId, workerName, "success", startedAt);
      logger.info(`[${workerName}] Completed`, { eventId: data.eventId });
    } else {
      await handleFailure(data, workerName, startedAt, result);
    }
  } catch (error) {
    const errMsg = error instanceof Error ? error.message : String(error);
    logger.error(`[${workerName}] Error`, { eventId: data.eventId, error: errMsg });
    await handleFailure(data, workerName, startedAt, { success: false, retryable: true, reason: errMsg });
    throw error; // Re-throw so pg-boss applies retry policy
  }
}

async function handleFailure(
  data: WorkerJobData,
  workerName: string,
  startedAt: Date,
  result: WorkerResult
): Promise<void> {
  const event = await prisma.event.findUnique({ where: { id: data.eventId } });
  if (!event) return;

  const nextRetryCount = event.retry_count + 1;
  const maxRetries = event.max_retries;

  if (!result.retryable || nextRetryCount > maxRetries) {
    // Dead-letter
    await prisma.event.update({
      where: { id: data.eventId },
      data: { status: "dead_lettered", retry_count: nextRetryCount },
    });
    await prisma.deadLetterEvent.create({
      data: {
        original_event_id: data.eventId,
        event_type: event.event_type,
        blueprint_id: data.blueprintId,
        user_id: data.userId,
        payload: event.payload as any,
        failure_reason: result.reason ?? "Unknown failure",
        retry_count: nextRetryCount,
        last_error: result.reason,
      },
    });
    await writeEventLog(data.eventId, workerName, "dead_lettered", startedAt, result.reason);
    logger.error(`[${workerName}] Dead-lettered`, { eventId: data.eventId, reason: result.reason });
  } else {
    // Retry
    await prisma.event.update({
      where: { id: data.eventId },
      data: { status: "retry_pending", retry_count: nextRetryCount },
    });
    await writeEventLog(data.eventId, workerName, "retry", startedAt, result.reason);
  }
}

async function writeEventLog(
  eventId: string,
  worker: string,
  result: string,
  startedAt: Date,
  errorMessage?: string
): Promise<void> {
  const finishedAt = new Date();
  await prisma.eventLog.create({
    data: {
      event_id: eventId,
      worker,
      result: result as any,
      error_message: errorMessage,
      duration_ms: finishedAt.getTime() - startedAt.getTime(),
      processed_at: finishedAt,
    },
  }).catch(() => null); // Non-critical — don't fail the job
}

  reason: z.string().min(1),
});

export async function POST(req: NextRequest, { params }: { params: { id: string } }) {
  try {
    await requireAuth(req, "INTERNAL_SERVICE");
    const body = await req.json();
    const { action, reason } = schema.parse(body);

    const dlEvent = await prisma.deadLetterEvent.findUnique({ where: { id: params.id } });
    if (!dlEvent) {
      return NextResponse.json(failure("NOT_FOUND", "Dead letter event not found"), { status: 404 });
    }

    await prisma.deadLetterEvent.update({
      where: { id: params.id },
      data: { resolved: true, reviewed: true, reviewed_at: new Date() },
    });

    return NextResponse.json(success({ resolved: true, action, reason }));
  } catch (error) {
    if (error instanceof TorgimaError) {
      return NextResponse.json(failure(error.code, error.message), { status: 403 });
    }
    return NextResponse.json(failure("INTERNAL_ERROR", "Failed to resolve dead letter"), { status: 500 });
  }
