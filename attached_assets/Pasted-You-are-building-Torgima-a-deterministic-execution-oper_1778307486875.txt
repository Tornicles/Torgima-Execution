pg-boss singleton. All background jobs go through this. Workers register here. Never create a PgBoss instance anywhere else.
import PgBoss from "pg-boss";

const globalForBoss = globalThis as unknown as {
  boss: PgBoss | undefined;
};

export function createBoss(): PgBoss {
  return new PgBoss({
    connectionString: process.env.DATABASE_URL!,
    max: 10,
    monitorStateIntervalSeconds: 30,
    deleteAfterDays: 7,
    retryLimit: 3,
    retryDelay: 30,
    retryBackoff: true,
    expireInHours: 24,
    onComplete: true,
  });
}

export async function getBoss(): Promise<PgBoss> {
  if (!globalForBoss.boss) {
    globalForBoss.boss = createBoss();
    await globalForBoss.boss.start();
  }
  return globalForBoss.boss;
}

src/lib/logger.ts
Structured logger. All engines use this. No raw console.log in production code.
type LogLevel = "debug" | "info" | "warn" | "error";

interface LogEntry {
  level: LogLevel;
  message: string;
  context?: Record<string, unknown>;
  timestamp: string;
}

function log(level: LogLevel, message: string, context?: Record<string, unknown>): void {
  const entry: LogEntry = {
    level,
    message,
    context,
    timestamp: new Date().toISOString(),
  };

  if (process.env.NODE_ENV === "development") {
    console[level === "debug" ? "log" : level](
      `[${entry.timestamp}] [${level.toUpperCase()}] ${message}`,
      context ?? ""
    );
  } else {
    console.log(JSON.stringify(entry));
  }
}

export const logger = {
  debug: (message: string, context?: Record<string, unknown>) => log("debug", message, context),
  info:  (message: string, context?: Record<string, unknown>) => log("info",  message, context),
  warn:  (message: string, context?: Record<string, unknown>) => log("warn",  message, context),
  error: (message: string, context?: Record<string, unknown>) => log("error", message, context),
};

src/types/api.ts
Standard response envelope. Every API route returns this shape.
export interface ApiResponse<T = unknown> {
  ok: boolean;
  data?: T;
  error?: {
    code: string;
    message: string;
    fieldErrors?: Record<string, string[]>;
    retryable?: boolean;
    currentStateVersion?: number;
  };
  meta?: {
    requestId?: string;
    timestamp?: string;
  };
}

export function success<T>(data: T, meta?: ApiResponse["meta"]): ApiResponse<T> {
  return { ok: true, data, meta };
}

export function failure(
  code: string,
  message: string,
  opts?: {
    fieldErrors?: Record<string, string[]>;
    retryable?: boolean;
    currentStateVersion?: number;
  }
): ApiResponse {
  return {
    ok: false,
    error: { code, message, ...opts },
  };
}

STEP 7 — WORKER REGISTRY
src/workers/WorkerRegistry.ts
All workers register here. This is the single place workers are started. Nothing registers workers directly — they all go through this registry.
import { getBoss } from "@/lib/queue";
import { logger } from "@/lib/logger";

export interface WorkerDefinition {
  name: string;
  queue: string;
  handler: (job: any) => Promise<void>;
  options?: {
    teamSize?: number;
    pollingIntervalSeconds?: number;
  };
}

class WorkerRegistry {
  private workers: WorkerDefinition[] = [];
  private started = false;

  register(worker: WorkerDefinition): void {
    this.workers.push(worker);
  }

  async startAll(): Promise<void> {
    if (this.started) return;
    const boss = await getBoss();

    for (const worker of this.workers) {
      await boss.work(
        worker.queue,
        worker.options ?? {},
        async (job) => {
          logger.info(`[Worker] Processing ${worker.queue}`, { jobId: job.id });
          try {
            await worker.handler(job);
            logger.info(`[Worker] Completed ${worker.queue}`, { jobId: job.id });
          } catch (error) {
            logger.error(`[Worker] Failed ${worker.queue}`, {
              jobId: job.id,
              error: error instanceof Error ? error.message : String(error),
            });
            throw error;
          }
        }
      );
      logger.info(`[Worker] Registered ${worker.name} on queue ${worker.queue}`);
    }

    this.started = true;
    logger.info(`[WorkerRegistry] All ${this.workers.length} workers started`);
  }
}

export const workerRegistry = new WorkerRegistry();

import { NextResponse } from "next/server";
