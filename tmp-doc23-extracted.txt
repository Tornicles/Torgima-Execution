TASK 1 — DECISION TRACE SERVICE
// DecisionTraceService.ts
// Called by every engine that makes a significant decision
// Writes to decision_traces table (exists from DOC 02 schema)
 
export class DecisionTraceService {
  static async write(input: DecisionTraceData): Promise<string> {
    // 1. Validate decision type is in approved list
    const validTypes = [&apos;USER_STRATEGIC&apos;,&apos;USER_EXECUTION&apos;,&apos;SYSTEM_CONTROL&apos;,
      &apos;SYSTEM_GENERATION&apos;,&apos;SYSTEM_CORRECTION&apos;,&apos;AI_ASSISTED&apos;,&apos;ADMIN_REPAIR&apos;];
    if (!validTypes.includes(input.decisionType)) throw new Error(&apos;INVALID_DECISION_TYPE&apos;);
 
    // 2. Validate required fields
    if (!input.decisionKey || !input.selectedOption) throw new Error(&apos;MISSING_REQUIRED_FIELDS&apos;);
 
    // 3. Write to decision_traces table
    const trace = await prisma.decisionTrace.create({ data: {
      user_id: input.userId, blueprint_id: input.blueprintId, order_id: input.orderId,
      decision_type: input.decisionType, decision_key: input.decisionKey,
      selected_option: input.selectedOption,
      options_considered: input.optionsConsidered,
      rejected_options: input.rejectedOptions ?? [],
      rationale_summary: input.rationaleSummary,
      evidence_ids: input.evidenceIds ?? [],
      source_engine: input.sourceEngine,
      confidence: input.confidence ?? &apos;MEDIUM&apos;,
      state_version: input.stateVersion,
      causation_event_id: input.causationEventId,
    }});
 
    // 4. If decision adds unresolved choices, trigger cognitive load update
    const addsChoices = input.optionsConsidered.length > 1 && !input.selectedOption;
    if (addsChoices) {
      await emitEvent({ event_type: &apos;cognitive_load_evaluation_requested&apos;,
        payload: { userId: input.userId, reason: &apos;new_unresolved_decision&apos; } });
    }
 
    return trace.id;
  }
 
  // Memory for Bhudhi context — returns last N decisions for a user/blueprint
  static async loadRecent(userId: string, blueprintId: string, limit = 5): Promise<DecisionTrace[]> {
    return prisma.decisionTrace.findMany({
      where: { user_id: userId, blueprint_id: blueprintId },
      orderBy: { created_at: &apos;desc&apos; }, take: limit,
    });
  }
}

// SIMPLIFICATION ACTIONS BY CONDITION:
 
// SIMPLIFY (MEDIUM load):
//   → frontend hides advanced options until user requests them
//   → Bhudhi explains next step more simply
//   → reduce optionsConsidered to 2 for any user-facing decision

//   → generate clarification task before next execution task
//   → reduce task content to: title + 1 sentence instruction + definition of done
