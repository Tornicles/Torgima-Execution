TASK 2 — REWARD ENGINE
// RewardEngine.ts
// Central service — called by RewardWorker when reward_evaluation_requested event fires
 
export async function evaluateReward(request: RewardEvaluationRequest): Promise<void> {
  // 1. Load source record and verify it exists and is owned by user
  const source = await loadSourceRecord(request.sourceType, request.sourceId, request.userId);
  if (!source) { await emitRewardBlocked(request, 'SOURCE_NOT_FOUND'); return; }
 
  // 2. Check idempotency — same source cannot create same reward twice
  const idempotencyKey = `${request.userId}:${request.sourceType}:${request.sourceId}:${rewardType}`;
  const existing = await findExistingReward(idempotencyKey);
  if (existing) { return; }  // already rewarded — silently return
 
  // 3. Determine which reward type this source event qualifies for
  const rewardType = await RewardEligibilityChecker.getEligibleRewardType(request.sourceType, source);
  if (!rewardType) { return; }  // not eligible for any reward
 
  // 4. Check blocking conditions
  const blocked = await checkBlockingConditions(request.userId, request.blueprintId, rewardType);
  if (blocked.isBlocked) {
    await emitEvent({ event_type: 'reward_blocked', payload: { reason: blocked.reason } });
    return;
  }
 
  // 5. Load condition for intensity selection
  const condition = await getCurrentCondition(request.blueprintId);
 
  // 6. Determine intensity
  const intensity = selectIntensity(rewardType, condition, source);
 
  // 7. Generate reward copy
  const copy = await generateRewardCopy(rewardType, intensity, source, condition);
 
  // 8. Create RewardEvent record
  await prisma.rewardEvent.create({ data: {
    user_id: request.userId, blueprint_id: request.blueprintId,
    source_type: request.sourceType, source_id: request.sourceId,
    reward_type: rewardType, intensity,
    title: copy.title, message: copy.message,
    visible_to_user: true, idempotency_key: idempotencyKey,
  }});
 
  // 9. Update streak metrics
  await StreakTracker.update(request.userId, request.blueprintId, rewardType);
 
  // 10. Emit downstream events
  await emitEvent({ event_type: 'reward_created', payload: { reward_type: rewardType, intensity } });
  if (rewardType === 'IDENTITY_ALIGNMENT') {
    await emitEvent({ event_type: 'identity_evaluation_requested', payload: { reason: 'reward_identity_alignment' } });
  }
  await emitEvent({ event_type: 'dashboard_reward_refresh_requested' });
}

TASK 3 — INTENSITY SELECTION
// selectIntensity() rules:
 
// LOW: normal completion, honest failure, simple return, small task
// MEDIUM: consistency reward, recovery reward, capability growth, focus protection
// HIGH: milestone, validated outcome, identity stage advancement
 
// CONDITION MODIFIERS:
// condition = 'danger' OR 'nothingness': never use HIGH intensity
//   → reduces to MEDIUM or LOW to avoid false pressure
// condition = 'affluence' OR 'power': HIGH intensity is allowed freely
 
// ANTI-ESCALATION RULE:
// If user received HIGH reward in last 24 hours: max intensity = MEDIUM for next reward
// Prevents intensity inflation from batch-completing tasks

TASK 4 — STREAK TRACKER
// StreakTracker.ts
// Maintains 5 streak metrics stored in a single user_streaks row or in bhudhi_memory
 
// STREAKS:
// execution_streak: consecutive calendar days with >= 1 accepted feedback
//   Reset to 0 if a full expected execution day is missed
//   NOT reset on protected gaps (streak_freeze from enforcement)
 
// return_streak: consecutive sessions where user returned to open task
//   Increments on any task start — does not require completion
//   Resets after 48h of no task start
 
// recovery_streak: consecutive recovery tasks completed with accepted feedback
//   Only counts tasks with type='recovery'
//   Resets when inactivity occurs without recovery
 
// truth_streak: consecutive feedback submissions that were honest
//   Includes FAILED_CONFIRMED, BLOCKED_CONFIRMED, and ACCEPTED
//   Resets if feedback is rejected for gaming or invalid proof
 
// focus_streak: consecutive completions of the highest-priority available task
//   Only tracked when PriorityEngine is active
//   Resets when lower-priority task is completed while higher exists
 
// STREAK FREEZE:
// enforcement_package.feature_permissions.streak_freeze = true
// → do not reset execution_streak even if day is missed
// → used during planned breaks or controlled recovery periods
 
// SHAME-FREE RULE:
// Never show 'streak lost' language in user-facing copy
// Instead show: 'Start your streak again — here is your first step'
// Streaks are motivators, not punishments
