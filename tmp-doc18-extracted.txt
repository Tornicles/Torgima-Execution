TASK 2 — IDENTITY STAGE EVALUATOR
// IdentityStageEvaluator.ts
// Called after: accepted feedback, signal update, pattern detection, milestone achieved
 
export async function evaluateIdentityStage(
  userId: string,
  blueprintId: string,
): Promise<IdentityEvaluationResult> {
 
  // 1. Load current identity state
  const currentState = await loadCurrentIdentityState(userId, blueprintId);
 
  // 2. Load evidence
  const evidence = await TransformationEvidenceCollector.collect(userId, blueprintId);
 
  // 3. Check for regression conditions first (always checked regardless of current stage)
  const regressionTriggered = (
    evidence.missed_windows_last_7_days >= 3 ||
    evidence.critical_false_progress_open === true
  );
 
  if (regressionTriggered && !['REGRESSING','UNFORMED'].includes(currentState.stage)) {
    return await advanceToStage(userId, blueprintId, 'REGRESSING', currentState, evidence);
  }
 
  // 4. Check for restart conditions (after regression)
  if (currentState.stage === 'REGRESSING' && evidence.recovery_tasks_completed >= 1) {
    return await advanceToStage(userId, blueprintId, 'RESTARTING', currentState, evidence);
  }
 
  // 5. Check advancement conditions in order
  const nextStage = determineNextStage(currentState.stage, evidence);
 
  if (nextStage && nextStage !== currentState.stage) {
    // 6. Check if any false progress flags block advancement
    const openFlags = await getOpenFalseProgressFlags(userId, blueprintId);
    const advancementBlocked = STAGES_BLOCKED_BY_FALSE_PROGRESS.includes(nextStage)
      && openFlags.length > 0;
 
    if (advancementBlocked) {
      return { evaluated: true, advanced: false, held: true,
        hold_reason: 'false_progress_flag_open', currentStage: currentState.stage };
    }
 
    return await advanceToStage(userId, blueprintId, nextStage, currentState, evidence);
  }
 
  return { evaluated: true, advanced: false, currentStage: currentState.stage };
}
 
const STAGES_BLOCKED_BY_FALSE_PROGRESS = ['BUILDING','STABILIZING','EXPANDING','INTEGRATED'];

TASK 3 — TRANSFORMATION EVIDENCE COLLECTOR
// TransformationEvidenceCollector.ts
// Gathers all evidence needed for stage evaluation
 
export async function collect(userId: string, blueprintId: string) {
  return {
    // Feedback cadence
    accepted_feedback_count_7d: await countAcceptedFeedback(userId, 7),
    accepted_feedback_count_14d: await countAcceptedFeedback(userId, 14),
    accepted_feedback_count_30d: await countAcceptedFeedback(userId, 30),
    consistency_7d: await calculateConsistency(userId, 7),
    consistency_14d: await calculateConsistency(userId, 14),
 
    // Recovery and resilience
    recovery_tasks_completed: await countRecoveryTasksCompleted(userId, blueprintId),
    returned_after_inactivity: await hasReturnedAfterInactivity(userId, blueprintId),
 
    // Difficulty and capability
    current_difficulty_level: await getCurrentDifficultyLevel(userId, blueprintId),
    capability_upgrades_proven: await countCapabilityUpgrades(userId, blueprintId),
 
    // Progress and outcomes
    milestones_completed: await countMilestonesCompleted(userId, blueprintId),
    outcomes_validated: await countValidatedOutcomes(userId, blueprintId),
 
    // False progress
    critical_false_progress_open: await hasCriticalFalseProgressFlag(userId, blueprintId),
    any_false_progress_open: await hasAnyFalseProgressFlag(userId, blueprintId),
 
    // Inactivity
    missed_windows_last_7_days: await countMissedWindows(userId, 7),
    days_since_last_activity: await getDaysSinceLastActivity(userId, blueprintId),
  };
}
