TASK 2 â€” TRIGGER ROUTER
// BhudhiTriggerRouter.ts
// The ONLY way to call Bhudhi in Phase 1
 
export interface BhudhiTriggerRequest {
  triggerType: keyof typeof REGISTERED_TRIGGERS;
  sourceEngine: string;
  userId: string;
  blueprintId?: string;
  stateVersion: number;
  input: Record<string, unknown>;
  idempotencyKey: string;
}
 
export class BhudhiTriggerRouter {
  async trigger(request: BhudhiTriggerRequest): Promise<BhudhiValidatedOutput> {
 
    // 1. Look up registered trigger
    const triggerDef = REGISTERED_TRIGGERS[request.triggerType];
    if (!triggerDef) throw new BhudhiRejected('UNREGISTERED_TRIGGER');
 
    // 2. Verify source engine is allowed
    if (!triggerDef.allowedSources.includes(request.sourceEngine))
      throw new BhudhiRejected('SOURCE_NOT_ALLOWED');
 
    // 3. Check budget
    const budgetOk = await BudgetController.checkAndReserve(request.userId);
    if (!budgetOk) return FallbackLibrary.get(request.triggerType);
 
    // 4. Build minimized context packet
    const contextPacket = ContextBuilder.build(
      request.input, triggerDef.requiredContext, triggerDef.forbiddenContext
    );
 
    // 5. Write BhudhiLog (pending)
    const logId = await BhudhiLogWriter.startLog({
      userId: request.userId, blueprintId: request.blueprintId,
      triggerType: request.triggerType, sourceEngine: request.sourceEngine,
      stateVersion: request.stateVersion,
    });
 
    try {
      // 6. Call provider
      const rawOutput = await ProviderAdapter.call({
        systemPrompt: buildSystemPrompt(triggerDef.mode, request.triggerType),
        userPrompt: JSON.stringify(contextPacket),
        maxTokens: triggerDef.maxTokens,
        timeoutMs: triggerDef.timeoutMs,
      });
 
      // 7. Parse and validate
      const validated = OutputValidator.validate(
        rawOutput, request.triggerType, request.stateVersion
      );
      if (!validated.ok) {
        await BhudhiLogWriter.failLog(logId, validated.errors);
        return FallbackLibrary.get(request.triggerType);
      }
 
      // 8. Complete log
      await BhudhiLogWriter.completeLog(logId, validated.output);
      return { ok: true, data: validated.output, logId };
 
    } catch (error) {
      await BhudhiLogWriter.failLog(logId, [error.message]);
      return FallbackLibrary.get(request.triggerType);
    }
  }
}
