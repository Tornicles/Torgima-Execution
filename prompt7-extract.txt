export interface BurnoutSignals {
  completionRateDrop: number    // This week vs last week

  sessionFrequencyDrop: number  // Login frequency decline
  consecutiveDeclines: number   // Signal decline streak
  inactivitySpikes: number      // Number of missed execution days
}

// SCORING RULES (deterministic formula):
// burnoutScore = weighted sum of signal factors
// Weights:
//   completionRateDrop &gt; 0.25:     +0.25
//   completionRateDrop &gt; 0.40:     +0.15 (additional)

//   reflectionLengthDrop &gt; 0.40:   +0.15
//   consecutiveDeclines &gt;= 5:      +0.15
//   inactivitySpikes &gt;= 3 in 7d:   +0.10
//
// Thresholds:
//   0.0-0.30: LOW    â€” monitor
