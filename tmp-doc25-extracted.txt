TASK 3 — FRICTION DETECTION FLOW
// FrictionDetectionEngine.ts
// Triggered after: FAILED/BLOCKED feedback, inactivity, failure classification,
// cognitive load spike, pattern detection
 
export async function detectFriction(context: FrictionContext): Promise<void> {
  // 1. Load all evidence (feedback, order, capability, condition, inactivity,
  //    patterns, cognitive load, dependencies, outcome validations)
  const evidence = await loadFrictionEvidence(context);
 
  // 2. Run FrictionClassifier — returns list of detected frictions with severity
  const detectedFrictions = await FrictionClassifier.classify(evidence);
  if (detectedFrictions.length === 0) return;
 
  // 3. Select primary friction type (highest severity, then highest confidence)
  const primary = detectedFrictions.sort(
    (a, b) => SEVERITY_ORDER[b.severity] - SEVERITY_ORDER[a.severity]
  )[0];
 
  // 4. Write BehaviorEvent for friction evidence
  await prisma.behaviorEvent.create({ data: {
    user_id: context.userId, blueprint_id: context.blueprintId,
    event_type: 'FRICTION_SPIKE', source_type: 'FEEDBACK',
    source_id: context.feedbackId,
    metadata: { friction_type: primary.type, severity: primary.severity },
  }});
 
  // 5. Check if repeated friction (same type 2+ times in 7 days)
  const isRepeated = await checkRepeatedFriction(context.userId, primary.type);
  if (isRepeated) {
    // Emit pattern_scan_requested to PatternLearningEngine
    await emitEvent({ event_type: 'pattern_scan_requested',
      payload: { reason: `repeated_${primary.type}` } });
  }
 
  // 6. Route adaptation
  await AdaptationRouter.route(primary.type, context);
}
 
const SEVERITY_ORDER = { LOW: 1, MEDIUM: 2, HIGH: 3, CRITICAL: 4 };
