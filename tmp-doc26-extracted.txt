TASK 2 — EXECUTABILITY CHECKER
// ExecutabilityChecker.ts
// Called by TaskGenerationEngine AND TaskGuaranteeEngine before publishing any task
 
export async function checkExecutability(
  candidateTask: CandidateTask,
  blueprintId: string,
  userId: string,
): Promise<ExecutabilityResult> {
 
  const blockingDependencies: BlockingDependency[] = [];
 
  // CHECK 1: SEQUENCE dependency
  // Does candidateTask.action_id have a prerequisite action that is not complete?
  const actionDeps = await loadProjectDependencies(candidateTask.action_id);
  for (const dep of actionDeps) {
    if (dep.status === 'OPEN' && dep.blocking === true) {
      blockingDependencies.push({
        type: 'SEQUENCE', dependencyId: dep.id,
        description: `Action requires ${dep.required_entity_id} to be complete first`,
        resolutionStrategy: 'complete_prerequisite_task',
      });
    }
  }
 
  // CHECK 2: CAPABILITY dependency
  // Does this task type require a capability the user has not proven?
  const capGap = await getOpenCapabilityGap(userId, blueprintId, candidateTask.task_type);
  if (capGap && capGap.blocks_execution) {
    blockingDependencies.push({
      type: 'CAPABILITY', dependencyId: capGap.id,
      description: `Requires ${capGap.capability_id} which is not yet proven`,
      resolutionStrategy: 'academy_task',
    });
  }
 
  // CHECK 3: SYSTEM_STATE dependency
  // Is there an open system health issue that blocks this blueprint?
  const healthLock = await getActiveHealthLock(blueprintId);
  if (healthLock) {
    blockingDependencies.push({
      type: 'SYSTEM_STATE', dependencyId: healthLock.id,
      description: 'System state requires repair before task can be published safely',
      resolutionStrategy: 'system_repair',
    });
  }
 
  // CHECK 4: ORDER-LEVEL dependency
  const orderDeps = await getOpenOrderDependencies(candidateTask.id);
  for (const dep of orderDeps.filter(d => d.blocking)) {
    blockingDependencies.push({ type: dep.dependency_type, dependencyId: dep.id,
      description: dep.resolution_strategy, resolutionStrategy: dep.resolution_strategy });
  }
 
  return {
    executable: blockingDependencies.length === 0,
    blockingDependencies,
    unlock_value: blockingDependencies.length === 0 ? 0 : calculateUnlockValue(blockingDependencies),
  };
}

TASK 4 — RESOLUTION TASK GENERATION
// DependencyResolver.ts
// When task is not executable: create the resolution task instead
 
async function generateResolutionTask(
  dependency: BlockingDependency,
  context: DependencyContext,
): Promise<void> {
 
  // Map dependency type to task type:
  const taskTypeMap = {
    SEQUENCE: 'blocker_resolution',
    DECISION: 'clarification',
    INFORMATION: 'clarification',
    ASSET: 'clarification',
    PERMISSION: 'blocker_resolution',
    EXTERNAL_RESPONSE: 'blocker_resolution',
    CAPABILITY: 'academy',  // CapabilityGapDetector handles this
    SYSTEM_STATE: 'system_repair',  // StateIntegrityEngine handles this
    OUTCOME_PROOF: 'phase_checkpoint',
  };
 
  // Emit the correct event for TaskGenerationEngine or the right engine:
  const event = dependency.type === 'CAPABILITY' ? 'erector_capability_gap_detected'
    : dependency.type === 'SYSTEM_STATE' ? 'system_repair_requested'
    : 'dependency_resolution_task_requested';
 
  await emitEvent({ event_type: event, payload: {
    dependency_id: dependency.dependencyId,
    dependency_type: dependency.type,
    resolution_strategy: dependency.resolutionStrategy,
    blocked_task_id: context.candidateTaskId,
  }});
}
