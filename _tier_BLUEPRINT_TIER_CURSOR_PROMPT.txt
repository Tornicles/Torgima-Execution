2.2 Create getTierConfig utility
export function getTierConfig(tier: BlueprintTier): TierConfig {
  return TIER_CONFIG[tier];
}

export function getTierConfigFromBlueprint(blueprint: Blueprint): TierConfig {
  // First try stored snapshot (guaranteed consistent)
  if (blueprint.tier_config && Object.keys(blueprint.tier_config).length &gt; 0) {
    return blueprint.tier_config as TierConfig;
  }
  // Fallback to live config if snapshot missing
  return TIER_CONFIG[blueprint.tier ?? "STARTER"];
}
