OBJECTIVE
Build the External Signal Integration system that processes user-uploaded proof files, submitted evidence links, and approved external webhooks. Classify each signal by source trust level, validate it against known patterns, store it as an ExternalSignal record, and route it to the correct downstream engine. External data is never automatically trusted — it always requires interpretation.

SURGICAL RULE
External signals feed the erector_evidence_ledger and the existing outcome validation system. They do NOT directly mutate blueprint state, signals, or conditions. Every external signal must be interpreted by the correct engine before any state change occurs. Bhudhi may summarize but cannot validate external truth.

TASK 2 — FILE UPLOAD PROCESSING
// POST /api/external/upload
// Auth: required. Accepts: multipart/form-data
// Allowed file types: pdf, png, jpg, jpeg, txt, csv, xlsx, docx
// Max file size: 10MB
 
async function processFileUpload(file: File, userId: string, blueprintId: string, context: UploadContext) {
  // 1. Validate file type and size
  if (!ALLOWED_TYPES.includes(file.type)) return reject('FILE_TYPE_NOT_ALLOWED');
  if (file.size > 10_000_000) return reject('FILE_TOO_LARGE');
 
  // 2. Store file in Supabase storage under /evidence/{userId}/{blueprintId}/
  const storagePath = await supabase.storage.uploadFile(file, userId, blueprintId);
 
  // 3. Score trust
  const trust = await ExternalTrustScorer.score({
    sourceType: 'user_upload', fileType: file.type, fileSize: file.size,
    uploadedBy: userId, context: context.description,
  });
 
  // 4. Submit to EvidenceLedgerService
  const evidence = await evidenceLedgerService.submit({
    userId, blueprintId,
    taskId: context.taskId, academyTaskId: context.academyTaskId,
    sourceType: 'user_upload', sourceLocation: storagePath,
    contentJson: { file_name: file.name, file_type: file.type, description: context.description },
    contentSummary: context.description?.substring(0, 200),
  });
 
  // 5. Emit external_signal_received event
  await emitEvent({ event_type: 'external_signal_received',
    payload: { evidence_id: evidence.evidence_id, trust_level: trust.level, signal_type: 'user_upload' } });
 
  return { evidence_id: evidence.evidence_id, trust_level: trust.level };
}

TASK 4 — WEBHOOK ROUTER
// POST /api/webhooks/:source
// :source must be in APPROVED_WEBHOOK_SOURCES list
// Requires: X-Webhook-Signature header for signature verification
 
async function processWebhook(source: string, payload: unknown, signature: string) {
  // 1. Verify source is approved
  if (!APPROVED_WEBHOOK_SOURCES.includes(source)) return reject(401);
 
  // 2. Verify signature
  const signatureValid = await verifyWebhookSignature(source, payload, signature);
  if (!signatureValid) {
    await writeSystemHealthLog('WARN', 'WEBHOOK_SIGNATURE_INVALID', { source });
    return reject(401);
  }
 
  // 3. Parse payload into ExternalSignal
  const signal = await parseWebhookPayload(source, payload);
  if (!signal) return reject(400, 'PAYLOAD_SCHEMA_INVALID');
 
  // 4. Check for duplicate (idempotency — same external_id from same source)
  const duplicate = await checkDuplicateWebhook(source, signal.external_id);
  if (duplicate) return ok({ duplicate: true });
 
  // 5. Route to correct engine based on signal type
  if (signal.signal_type === 'payment_confirmed') {
    await evidenceLedgerService.submit({ sourceType: 'payment_signal', ... });
    await emitEvent({ event_type: 'outcome_validation_requested', ... });
  } else if (signal.signal_type === 'external_confirmation') {
    await emitEvent({ event_type: 'dependency_resolution_requested', ... });
  }
}
