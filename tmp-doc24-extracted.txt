TASK 2 — PRIORITY ENGINE
// PriorityEngine.ts
// Called by TaskGuaranteeEngine when multiple candidate tasks exist
// Also runs as validation after TaskGenerationEngine when DEBUG_PRIORITY flag is on
 
export async function selectHighestValueTask(
  candidateTasks: CandidateTask[],
  context: PriorityContext,
): Promise<PriorityResult> {
 
  if (candidateTasks.length === 0) throw new Error(&apos;NO_CANDIDATES&apos;);
  if (candidateTasks.length === 1) {
    return { selectedTask: candidateTasks[0], reason: &apos;single_candidate&apos;, score: 0 };
  }
 
  // 1. Check dependency executability for each candidate
  const executable = await filterExecutableCandidates(candidateTasks, context.blueprintId);
  if (executable.length === 0) {
    // All blocked — return dependency resolution task
    return { selectedTask: null, reason: &apos;all_blocked&apos;, requiresDependencyResolution: true };
  }
 
  // 2. Score each executable candidate
  const scored = await Promise.all(
    executable.map(async task => ({
      task,
      score: await TaskValueScorer.score(task, context),
    }))
  );
 
  // 3. Sort by score descending
  scored.sort((a, b) => b.score - a.score);
  const selected = scored[0];
 
  // 4. Apply override rules (override scoring when control is absolute)
  const overrideTask = await checkOverrideRules(executable, context);
  if (overrideTask) {
    return { selectedTask: overrideTask, reason: &apos;override_rule&apos;, score: 100 };
  }
 
  // 5. Write DecisionTrace for selection when multiple candidates competed
  await DecisionTraceService.write({
    decisionType: &apos;SYSTEM_GENERATION&apos;, decisionKey: &apos;priority_selection&apos;,
    selectedOption: selected.task.id,
    optionsConsidered: scored.map(s => `${s.task.id}(${s.score.toFixed(0)})`),
    rationaleSummary: `Selected highest value task. Score: ${selected.score.toFixed(0)}/100`,
    confidence: &apos;HIGH&apos;,
  });
 
  return { selectedTask: selected.task, score: selected.score, reason: &apos;value_score&apos; };
}
