LAW 1 — READ BEFORE TOUCHING. Read the Drizzle schema file completely before adding tables. Read every existing engine file before creating new engines. Read TriggerRouter.ts before adding Bhudhi calls. Read stateMachine.ts before adding enums. No exceptions.
LAW 2 — EXTEND, NEVER REPLACE. The existing validations table gets new columns added via Drizzle migration. The existing onboarding flow gets new screens inserted. The existing validation route gets upgraded. Nothing existing is deleted.
LAW 3 — ALL SCHEMA CHANGES GO THROUGH DRIZZLE MIGRATIONS. Never use runtime schema patching (ensureSchema was deleted in R1). Every new table and column gets a drizzle-kit generate migration. Run npx drizzle-kit check after applying. Zero drift is the standard.
LAW 4 — ALL BHUDHI CALLS GO THROUGH TriggerRouter. Every Anthropic API call in the Erector engines must call TriggerRouter.route() with a registered trigger type, mode, stateVersion, and idempotencyKey. Direct anthropic.messages.create calls are forbidden.
LAW 5 — RULE ENGINE SCORES. BHUDHI NARRATES. The scoring engine (Step 9) calculates all 11 scores deterministically without any AI involvement. Bhudhi provides narrative context and structured analysis. Bhudhi never sets a score value. Bhudhi never decides if a blueprint can launch.
LAW 6 — ERECTOR VALIDATION ROWS USE validation_kind = &apos;erector&apos;. The validations table was repaired in R2 to prevent payload type confusion. All Erector validation rows must write validation_kind = &apos;erector&apos;. LAW 7 — STATE MUTATIONS GO THROUGH StateMutationService. Any engine that updates blueprint state must use StateMutationService.mutate() from G4 — not direct db.update(). This maintains state_version integrity and writes blueprint_versions audit records.
LAW 8 — ALL STATE STRINGS FROM stateMachine.ts. Never hardcode &apos;active&apos;, &apos;pending&apos;, &apos;erector&apos;, &apos;starter&apos;, &apos;growth&apos;, &apos;ultimate&apos; as string literals. Import from src/types/stateMachine.ts. Add ErectorValidationStage and BlueprintLevel to that file.
LAW 9 — ALL EVENT NAMES FROM eventContracts.ts. Never hardcode &apos;erector_validation_complete&apos; or &apos;path_selected&apos; as strings. Add them to src/types/eventContracts.ts. Import everywhere.
LAW 10 — IDEMPOTENCY ENFORCED. Every validation run checks idempotency_key before processing. Duplicate submissions return the existing result. Every emitEvent call for Erector events includes idempotency_key.
LAW 11 — NO PROMISES. No language in any UI copy or Bhudhi output may promise a specific outcome. Use: probability, estimated, likely, projected. Never: guaranteed, will achieve, you will.
LAW 12 — EXECUTION LANGUAGE THROUGHOUT. No motivational fluff. No &apos;Great goal!&apos;, &apos;Amazing choice!&apos;, &apos;You&apos;ve got this!&apos;. Direct operational language only. // src/services/validation/scoringEngine.ts
// NO Bhudhi calls in this file. Zero. All math.
export class ScoringEngine {
calculateScores(engines: {
archetype: ArchetypeResolutionOutput;
purpose: PurposeDecompositionOutput;
industry: IndustryIntelligenceOutput;
market: MarketAnalysisOutput;
competitors: CompetitorAnalysis[];
executionDifficulty: ExecutionDifficultyOutput;
customerPain: CustomerPainOutput;
timing: TimingEngineOutput;
}, input: ErectorValidationInput): ErectorScores {
// Each score is 0-10. Apply archetype adjustment to relevant scores.
const adj = engines.archetype.primaryArchetype.blueprintDifficultyAdjustment;
const marketScore       = this.scoreMarket(engines.market) + adj;
const executionScore    = this.scoreExecution(engines.executionDifficulty, input.commitmentLevel);
const timingScore       = this.scoreTiming(engines.timing, engines.industry);
const competitionScore  = this.scoreCompetition(engines.competitors, engines.market);
const scalabilityScore  = this.scoreScalability(engines.market, engines.archetype);
const localEconomyScore = this.scoreLocalEconomy(engines.archetype.primaryArchetype);
const resourceScore     = this.scoreResources(input, engines.archetype);
const blueprintDiffScore = this.scoreBlueprintDifficulty(engines.executionDifficulty, engines.archetype);
const riskScore         = this.scoreRisk(engines);
const momentumPotential = this.scoreMomentum(engines.industry, engines.market);
const trustBuildingScore = this.scoreTrust(engines.archetype, engines.competitors);
const clamp = (v: number) =&gt; Math.max(0, Math.min(10, v));
const scores = {
marketScore: clamp(marketScore),
executionScore: clamp(executionScore),
timingScore: clamp(timingScore),
competitionScore: clamp(competitionScore),
scalabilityScore: clamp(scalabilityScore),
localEconomyScore: clamp(localEconomyScore),
resourceScore: clamp(resourceScore),
blueprintDifficultyScore: clamp(blueprintDiffScore),
riskScore: clamp(riskScore),
momentumPotential: clamp(momentumPotential),
trustBuildingScore: clamp(trustBuildingScore),
};
// Weighted composite Reality Score
const realityScore = clamp(
scores.marketScore * 0.20 +
scores.executionScore * 0.15 +
scores.timingScore * 0.10 +
scores.competitionScore * 0.10 +
scores.scalabilityScore * 0.10 +
scores.localEconomyScore * 0.15 +
scores.resourceScore * 0.10 +
scores.riskScore * 0.05 +
scores.momentumPotential * 0.05
);
// Deterministic verdict thresholds
const verdict = realityScore &gt;= 6.0 ? &apos;proceed&apos; : realityScore &gt;= 3.5 ? &apos;adjust&apos; : &apos;pivot&apos;;
// Blueprint level recommendation
const base = realityScore &gt;= 7.0 ? &apos;ultimate&apos; : realityScore &gt;= 5.0 ? &apos;growth&apos; : &apos;starter&apos;;
const level = this.applyBlueprintAdjustments(base, input, engines.archetype);
return { ...scores, realityScore, verdict, recommendedBlueprintLevel: level };
}
private applyBlueprintAdjustments(base: string, input: ErectorValidationInput, archetype: ArchetypeResolutionOutput): &apos;starter&apos; | &apos;growth&apos; | &apos;ultimate&apos; {
let level = base;
if (input.commitmentLevel === &apos;LOW&apos;) level = &apos;starter&apos;;
if (archetype.primaryArchetype.blueprintDifficultyAdjustment &gt; 1.5 &amp;&amp; level === &apos;ultimate&apos;) level = &apos;growth&apos;;
return level as &apos;starter&apos; | &apos;growth&apos; | &apos;ultimate&apos;;
}
}

// src/services/validation/erectorOrchestrationService.ts
export class ErectorOrchestrationService {
async run(input: ErectorValidationInput): Promise&lt;{ validationId: string; scores: ErectorScores; paths: {...}; }&gt; {
// Check idempotency
const existing = await this.checkIdempotency(input.idempotencyKey);
if (existing) return existing;
// Create validation row with validation_kind = &apos;erector&apos;
const validation = await db.insert(validations).values({
id: crypto.randomUUID(),
blueprint_id: input.blueprintDraftId,
user_id: input.userId,
validation_kind: &apos;erector&apos;,          // R2 requirement
contract_id: crypto.randomUUID(),     // stable reference for launch
status: &apos;running&apos;,
erector_validation_stage: ErectorValidationStage.RUNNING,
payload_version: 1,
// Intake fields:
location_country: input.location.country,
budget_available: input.budgetAvailable,
commitment_level: input.commitmentLevel,
// ... other intake fields
}).returning().then(r =&gt; r[0]);
// Step 1: Archetype resolution (required — all other engines depend on it)
const archetype = await archetypeResolutionEngine.resolve(input);
// Steps 2-8: Engines run in dependency order
const purpose     = await purposeDecompositionEngine.run(input, archetype);
const industry    = await industryIntelligenceEngine.run(input, archetype);
const market      = await marketAnalysisEngine.run(input, archetype, industry);
const competitors = await competitorRealityEngine.run(input, archetype, industry);
const economy     = await economyBriefingEngine.run(input, archetype);
const pain        = await customerPainEngine.run(input, archetype);
const difficulty  = await executionDifficultyEngine.run(input, archetype);
const simulations = await timeSimulationEngine.run(input, archetype, difficulty);
// Step 9: Scoring (deterministic)
const scores = scoringEngine.calculateScores({ archetype, purpose, industry, market, competitors, executionDifficulty: difficulty, customerPain: pain, timing: industry }, input);
// Step 10: Path generation (Bhudhi-assisted)
const paths = await pathEngine.generate(input, archetype, scores, simulations);
// Save complete Erector payload
await db.update(validations).set({
erector_payload: { archetype, purpose, industry, market, competitors, economy, pain, difficulty, simulations, paths },
erector_validation_stage: ErectorValidationStage.COMPLETE,
status: &apos;complete&apos;,
reality_score: scores.realityScore,
recommended_blueprint_level: scores.recommendedBlueprintLevel,
...flattenScores(scores),
}).where(eq(validations.id, validation.id));
// Emit event — using EventTypes from eventContracts.ts
await emitEvent({
event_type: EventTypes.ERECTOR_VALIDATION_COMPLETE,
entity_type: &apos;validation&apos;, entity_id: validation.id,
user_id: input.userId, blueprint_id: input.blueprintDraftId,
idempotency_key: `erector_complete:${validation.id}`,
payload: { contract_id: validation.contract_id, reality_score: scores.realityScore, verdict: scores.verdict },
});
// Mutate blueprint state via StateMutationService
const blueprint = await db.select().from(blueprints).where(eq(blueprints.id, input.blueprintDraftId)).limit(1).then(r=&gt;r[0]);
if (blueprint) {
await stateMutationService.mutate({
blueprintId: input.blueprintDraftId, userId: input.userId,
expectedVersion: blueprint.state_version,
mutationName: &apos;erector_validation_complete&apos;,
mutationFn: async (tx) =&gt; {
await tx.update(blueprints).set({ status: BlueprintStatus.PATH_SELECTION }).where(eq(blueprints.id, input.blueprintDraftId));
}
});
}
return { validationId: validation.id, scores, paths };
}
}
